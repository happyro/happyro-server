cmd_fild07,299,83,4	script	快艇船员#cmd::泽因#cmd	100,{
	set .@n$,"["+strnpcinfo(1)+"]";
	mes .@n$;
	mes "要乘坐瑞德鲁斯航线的船吗？";
	mes "你可以乘瑞德鲁斯前往艾尔贝塔或依斯鲁得。";
	next;
	switch(select("艾尔贝塔 - 600 Zeny:依斯鲁得 - 800 Zeny:取消")) {
	case 1:
		if (Zeny < 600) break;
		set Zeny, Zeny-600;
		warp "alberta",192,169;
		end;
	case 2:
		if (Zeny < 800) break;
		set Zeny, Zeny-800;
		if(checkre(0))
			warp "izlude",195,212;
		else
			warp "izlude",176,182;
		end;
	case 3:
		mes .@n$;
		mes "乘船仍是最安全可靠的交通方式之一。";
		mes "欢迎你快来体验瑞德鲁斯航线~";
		close;
	}
	mes .@n$;
	mes "抱歉，你没有足够的Zeny支付乘船费用。";
	close;
}
