payon,173,141,4	script	健身教练卡布斯#GymPass	899,{
	mes "[卡布斯]";
	mes "嘿，朋友。人们如今不像过去那样积极锻炼身体。即使您靠战斗为生，身体某些部位也可能虚弱松弛。";
	mes "您明白我的意思吗？";
	mes "";
	next;
	mes "[卡布斯]";
	mes "跟我训练吧，我保证您能举起并携带更多物品。";
	mes "每次只要给我^FF0000健身房通行证^000000，我们就可以开始了。";
	mes "";
	next;
	mes "[卡布斯]";
	mes "但别太兴奋：锻炼也可能过度。听说过有人练得太壮吗？";
	mes "因为他们都死了。明白？";
	mes "";
	next;
	mes "[卡布斯]";
	mes "我认为，跟我认真训练，可以安全地提升负重能力，最多训练 ^FF000010 次^000000。";
	mes "那么……准备好流汗了吗？";
	next;
	switch(select("是:否:我的训练效果消失了。")) {
	case 1:
	if (gympassmemory < 10) {
		set .@add_carry,gympassmemory + 1;
			set .@remain_carry,10 - .@add_carry;
			if (countitem(7776) > 0) {
			mes "[卡布斯]";
				mes "哦，太好了，我看到您带来了健身房通行证。";
				mes "好，照我做，感受肌肉的燃烧吧。";
				mes "准备好了吗？开始吧。";
				next;
				specialeffect2 EF_EARTHSPIKE;
				next;
				specialeffect2 EF_DEVIL;
				next;
				specialeffect2 EF_COIN;
				next;
				specialeffect2 EF_SIGHTRASHER;
				next;
				mes "[卡布斯]";
				mes "好了，现在您应该能携带更多东西了。";
				mes "让我看看，您还能继续这样训练 ^FF0000" + .@remain_carry + "^000000 次，提升负重能力。";
				delitem 7776,1; //Max_Weight_Up_Scroll
				set gympassmemory,.@add_carry;
				skill "ALL_INCCARRY",.@add_carry,SKILL_PERM_GRANT;
				close;
				}
			else {
			mes "[卡布斯]";
				mes "朋友，我怎么说的？";
				mes "想训练并增强负重能力，就必须带来您的 ^FF0000健身房通行证^000000。";
				mes "";
				close;
				}
			}
		else {
		mes "[卡布斯]";
			mes "朋友，我觉得您的负重能力不能再增强了。";
			mes "继续尝试对身体太危险！";
			mes "好了，我已经告诉过您极限。";
			mes "";
			close;
			}
		case 2:
	mes "[卡布斯]";
		mes "啊，真遗憾。";
		mes "改变主意后再回来吧。";
		mes "也可以告诉您的朋友：如果他们身材松弛，我会帮他们恢复体型。";
		mes "";
		close;
		case 3:
	if (gympassmemory > 0) {
		mes "[卡布斯]";
			mes "发生了什么？";
			mes "您的肌肉萎缩，负重能力下降了吗？幸运的是，肌肉有记忆，重新练回来不会花太久……";
			mes "";
			next;
			specialeffect2 EF_EARTHSPIKE;
			next;
			specialeffect2 EF_DEVIL;
			next;
			specialeffect2 EF_COIN;
			next;
			specialeffect2 EF_SIGHTRASHER;
			next;
			mes "[卡布斯]";
			mes "怎么样？";
			mes "您的负重能力就这样恢复了！";
			mes "下次别再偷懒了，好吗？";
			mes "";
			skill "ALL_INCCARRY",gympassmemory,SKILL_PERM_GRANT;
			close;
			}
		else {
		mes "[卡布斯]";
			mes "呃……";
			mes "我们以前没有一起训练过。";
			mes "我很确定。";
			close;
			}
		}
	}
