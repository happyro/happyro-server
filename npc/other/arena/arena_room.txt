function	script	Func_Are_Rew	{
	.@item_id = getarg(0);
	.@amount = getarg(1);
	.@arena_point = getarg(2);

	mes "[吉武]";
	mes "你想用竞技场点数兑换"+ getitemname(.@item_id) +"吗？";
	mes "每次用^3131FF"+ .@arena_point +"点竞技场点数兑换"+ .@amount +"个"+ getitemname(.@item_id) +"^000000。";
	mes "取消请输入0；要兑换，请输入兑换^3131FF次数^000000。";
	next;
	input .@reward;
	if (.@reward <= 0) {
		mes "[吉武]";
		mes "你取消了请求。";
		close;
	} else if (.@reward > 1500) {
		mes "[吉武]";
		mes "兑换次数超过上限，最多可兑换1,500次。";
		close;
	}
	if (arena_point < (.@reward * .@arena_point)) {
		mes "[吉武]";
		mes "你的竞技场点数不足。";
		mes "请先确认自己持有的竞技场点数。";
		close;
	}
	arena_point = arena_point - (.@reward * .@arena_point);
	getitem .@item_id,.@reward * .@amount;
	mes "[吉武]";
	mes "谢谢惠顾，欢迎再来。";
	close;
}
