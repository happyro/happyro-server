prt_are_in,103,11,3	script	竞技场点数管理员	732,{
	mes "[竞技场点数管理员]";
	mes "希望你在竞技场玩得开心。";
	mes "如果你在艾尔帕兰加速赛道获得了点数，我可以将其兑换为竞技场点数。";
	next;
	switch(select("点数查询:兑换点数:^660000兑换说明^000000")) {
	case 1:
		mes "[竞技场点数管理员]";
		mes strcharinfo(0) + "，你目前拥有：";
		mes arena_point + " 点竞技场点数，";
		mes tt_point + " 点加速赛道点数。";
		close;
	case 2:
		mes "[竞技场点数管理员]";
		mes strcharinfo(0) + "，你目前拥有：";
		mes arena_point + " 点竞技场点数，";
		mes tt_point + " 点加速赛道点数。";
		next;
		mes "[竞技场点数管理员]";
		mes "请选择加速赛道点数兑换为竞技场点数的方式。";
		mes "兑换10点以上时，只能以^4D4DFF10的倍数^000000兑换。";
		next;
		switch(select("2点赛道点数换1点竞技场点数:4点赛道点数换2点竞技场点数:6点赛道点数换3点竞技场点数:8点赛道点数换4点竞技场点数:兑换10点及以上:取消")) {
		case 1:
			callsub S_ExchangePoints,28999,2,1;
		case 2:
			callsub S_ExchangePoints,28998,4,2;
		case 3:
			callsub S_ExchangePoints,28997,6,3;
		case 4:
			callsub S_ExchangePoints,28996,8,4;
		case 5:
			mes "[竞技场点数管理员]";
			mes "每次将10点加速赛道点数兑换为5点竞技场点数。";
			mes "请输入兑换次数，最多20次。";
			mes "取消请输入“^3355FF0^000000”。";
			next;
			input .@input;
			if (.@input <= 0) {
				mes "[竞技场点数管理员]";
				mes "你已取消兑换。";
				close;
			}
			else if (.@input > 20) {
				mes "[竞技场点数管理员]";
				mes "兑换次数超过上限，请输入不超过20的数值。";
				close;
			}
			else {
				set .@want_point1,10 * .@input;
				set .@want_point,5 * .@input;
				set .@my_arena_all,arena_point + .@want_point;
				set .@my_turbo_all,tt_point - .@want_point1;
				if (.@my_arena_all > 28999) {
					mes "[竞技场点数管理员]";
					mes "此次兑换会超出竞技场点数的持有上限。";
					next;
					mes "[竞技场点数管理员]";
					mes "请先消费部分竞技场点数，再来使用此服务。";
					mes "感谢惠顾。";
					close;
				}
				if (.@my_turbo_all < 0) {
					mes "[竞技场点数管理员]";
					mes "抱歉，你的加速赛道点数不足，无法进行此次兑换。";
					close;
				}
				else {
					mes "[竞技场点数管理员]";
					mes "你已完成 " + .@input + " 次兑换，每次用10点加速赛道点数兑换5点竞技场点数。";
					mes "共消耗 " + .@want_point1 + " 点加速赛道点数，获得 " + .@want_point + " 点竞技场点数。";
					set tt_point, .@my_turbo_all;
					set arena_point, .@my_arena_all;
					next;
					mes "[竞技场点数管理员]";
					mes strcharinfo(0) + "，你现在拥有：";
					mes "^4682B4" + arena_point + "^000000 点竞技场点数，";
					mes "^00688B" + tt_point + "^000000 点加速赛道点数。";
					mes "感谢惠顾。";
					close;
				}
			}
		case 6:
			mes "[竞技场点数管理员]";
			mes "你已取消兑换。";
			close;
		}
	case 3:
		// Again. no dialog for this option.
		close;
	}
	close;

S_ExchangePoints:
	if (arena_point > getarg(0) ) {
		mes "[竞技场点数管理员]";
		mes "此次兑换会超出竞技场点数的持有上限。";
		next;
		mes "[竞技场点数管理员]";
		mes "请先消费部分竞技场点数，再来使用此服务。";
		mes "感谢惠顾。";
		close;
	}
	.@track_point = getarg(1);
	if (tt_point >= .@track_point) {
		.@arena_point = getarg(2);
		mes "[竞技场点数管理员]";
		mes strcharinfo(0) + "，兑换完成。";
		mes "你已将 " + .@track_point + " 点加速赛道点数兑换为 " + .@arena_point + " 点竞技场点数。";
		tt_point = tt_point - .@track_point;
		arena_point = arena_point + .@arena_point;
		next;
		mes "[竞技场点数管理员]";
		mes strcharinfo(0) + "，你现在拥有：";
		mes "^4682B4" + arena_point + "^000000 点竞技场点数，";
		mes "^00688B" + tt_point + "^000000 点加速赛道点数。";
		mes "感谢惠顾。";
		close;
	}
	else {
		mes "[竞技场点数管理员]";
		mes "抱歉，你没有足够的加速赛道点数。";
		mes "使用此兑换服务至少需要2点加速赛道点数。";
		close;
	}
}
