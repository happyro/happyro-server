	case 2:
		switch(select("炽焰球:闪电球:冰冻球:毒球:致盲球:取消")) {
		case 1: callsub S_Sphere_Exchange,13223,13228;
		case 2: callsub S_Sphere_Exchange,13224,13229;
		case 3: callsub S_Sphere_Exchange,13227,13230;
		case 4: callsub S_Sphere_Exchange,13225,13231;
		case 5: callsub S_Sphere_Exchange,13226,13232;
		case 6: callsub S_Close;
		}
	case 3:
		switch(select("旧式火焰弹:旧式闪电弹:旧式冰冻弹:旧式毒性弹:旧式暗盲弹:取消")) {
		case 1: callsub S_Old_Sphere_Exchange,13203,13228;
		case 2: callsub S_Old_Sphere_Exchange,13204,13229;
		case 3: callsub S_Old_Sphere_Exchange,13207,13230;
		case 4: callsub S_Old_Sphere_Exchange,13205,13231;
		case 5: callsub S_Old_Sphere_Exchange,13206,13232;
		case 6: callsub S_Close;

	case 4:
		switch(select("重弹L:重弹M:重弹H:霰弹弹药SH:霰弹弹药XH:取消")) {
		case 1: callsub S_Slug_Exchange,13210,24;
		case 2: callsub S_Slug_Exchange,13211,12;

S_Casing:
	.@bullet_id = getarg(0);
	.@pack_id = getarg(1);
	.@bullet_str$ = getitemname(.@bullet_id);
	.@pack_str$ = getitemname(.@pack_id);
	.@bullet_count = countitem(.@bullet_id);
	.@pack_count = .@bullet_count / 500;

	if (.@bullet_count < 500 || getequipid(EQI_AMMO) != -1 || BaseClass != Job_Gunslinger) {
		mes "[肯尼]";
		mes "嘿～这项兑换只对神枪手系职业开放，请准备好材料，并先卸下弹药栏中的所有弹药。";
		mes "你至少需要 500 发^FF0000" + .@bullet_str$ + "^000000，明白吗？兑换前请确保弹药栏^FF0000未装备任何弹药^000000。";
		close;
	}

	mes "[肯尼]";
	mes "你选择了^FF0000" + .@pack_str$ + "^000000。";
	mes "需要 500 发^3131FF" + .@bullet_str$ + "^000000才能制作 1 个^3131FF" + .@pack_str$ + "^000000，费用为^3131FF500^000000 Zeny。";
	mes "你想兑换多少？";
	next;
	switch(select("全部兑换:只制作1个:取消")) {
		case 1:
			mes "[肯尼]";
			mes "子弹数量：^3131FF" + .@bullet_count;
			mes "弹药筒数量：^3131FF" + .@pack_count;
			mes "所需 Zeny：^3131FF" + .@pack_count * 500;
			next;
			mes "要兑换吗？";
			next;
			switch(select("兑换:取消")) {
				case 1:
					if (Zeny < .@pack_count * 500) {
						mes "[肯尼]";
						mes "嘿～你的 Zeny 不够，请准备好费用再来。";
						mes "你必须有^FF0000" + .@pack_count * 500 + "^000000，明白吗？";
						close;
					}
					mes "[肯尼]";
					mes "完成了。";
					mes "打开弹药筒前，请预留足够的负重和背包空间。";
					Zeny -= .@pack_count * 500;
					delitem .@bullet_id, .@pack_count * 500;
					getitem .@pack_id, .@pack_count;
					close;
				case 2:
					close;
			}
		case 2:
			if (Zeny < 500) {
				mes "[肯尼]";
				mes "嘿～你的 Zeny 不够，请准备好费用再来。";
				mes "你需要 ^FF0000500^000000 Zeny，明白吗？";
				close;
			}
			mes "[肯尼]";
			mes "完成了。";
			mes "打开弹药筒前，请预留足够的负重和背包空间。";
			Zeny -= 500;
			delitem .@bullet_id, 500;
			getitem .@pack_id, 1;
			close;
		case 3:
			callsub S_Close;
	}

S_Sphere_Exchange:
	.@old_id = getarg(0);
	.@new_id = getarg(1);
	.@old_str$ = getitemname(.@old_id);
	.@new_str$ = getitemname(.@new_id);
	.@old_count = countitem(.@old_id);
	.@new_count = .@old_count / 10;

	if (.@old_count < 10 || getequipid(EQI_AMMO) != -1 || BaseClass != Job_Gunslinger) {
		mes "[肯尼]";
		mes "嘿～这项兑换只对神枪手系职业开放，请准备好材料，并先卸下弹药栏中的所有弹药。";
		mes "你至少需要 10 个^FF0000" + .@old_str$ + "^000000，明白吗？兑换前请确保弹药栏^FF0000未装备任何弹药^000000。";
		close;
	}

	mes "[肯尼]";
	mes "你选择了^FF0000" + .@old_str$ + "^000000。";
	mes "需要10个^3131FF" + .@old_str$ + "^000000才能制作1个^3131FF" + .@new_str$ + "^000000。";
	mes "你想兑换多少？";
	next;
	switch(select("全部兑换:只制作1个:取消")) {
		case 1:
			mes "[肯尼]";
			mes .@old_str$ + "数量：^3131FF" + .@old_count;
			mes .@new_str$ + "数量：^3131FF" + .@new_count;
			next;
			mes "要兑换吗？";
			next;
			switch(select("兑换:取消")) {
				case 1:
					mes "[肯尼]";
					mes "完成了。";
					delitem .@old_id, .@new_count * 10;
					getitem .@new_id, .@new_count;
					close;
				case 2:
					close;
			}
		case 2:
			mes "[肯尼]";
			mes "完成了。";
			delitem .@old_id, 10;
			getitem .@new_id, 1;
			close;
		case 3:
			callsub S_Close;
	}

S_Old_Sphere_Exchange:
	.@old_id = getarg(0);
	.@new_id = getarg(1);
	.@old_str$ = getitemname(.@old_id);
	.@new_str$ = getitemname(.@new_id);
	.@old_count = countitem(.@old_id);

	if (.@old_count < 1 || getequipid(EQI_AMMO) != -1 || BaseClass != Job_Gunslinger) {

	mes "[肯尼]";
	mes "你选择了^FF0000" + .@old_str$ + "^000000。";
	mes "需要1个^3131FF" + .@old_str$ + "^000000才能制作1个^3131FF" + .@new_str$ + "^000000。";
	mes "你想兑换多少？";
	next;
	switch(select("全部兑换:只制作1个:取消")) {
		case 1:
			mes "[肯尼]";
			mes .@old_str$ + "数量：^3131FF" + .@old_count;
			mes .@new_str$ + "数量：^3131FF" + .@old_count;
			next;
			mes "要兑换吗？";
			next;
			switch(select("兑换:取消")) {

					delitem .@old_id, .@old_count;
					getitem .@new_id, .@old_count;

				case 2:
					close;
			}
		case 2:
			mes "[肯尼]";
			mes "完成了。";
			delitem .@old_id, 1;
			getitem .@new_id, 1;
			close;
		case 3:
			callsub S_Close;
	}

S_Slug_Exchange:
	.@slug_id = getarg(0);
	.@req_count = getarg(1);
	.@slug_str$ = getitemname(.@slug_id);
	.@new_slug_str$ = getitemname(25187);
	.@count = countitem(.@slug_id);
	.@new_count = .@count / .@req_count;

	if (.@count < .@req_count || getequipid(EQI_AMMO) != -1 || BaseClass != Job_Gunslinger) {
		mes "[肯尼]";
		mes "嘿～这项兑换只对神枪手系职业开放，请准备好材料，并先卸下弹药栏中的所有弹药。";
		mes "你至少需要 " + .@req_count + " 个^FF0000" + .@slug_str$ + "^000000，明白吗？兑换前请确保弹药栏^FF0000未装备任何弹药^000000。";
		close;
	}

	mes "[肯尼]";
	mes "你选择了^FF0000" + .@slug_str$ + "^000000。";
	mes "需要" + .@req_count + "个^3131FF" + .@slug_str$ + "^000000才能制作1个^3131FF" + .@new_slug_str$ + "^000000。";
	mes "你想兑换多少？";
	next;
	switch(select("全部兑换:只制作1个:取消")) {
		case 1:
			mes "[肯尼]";
			mes .@slug_str$ + "数量：^3131FF" + .@count;
			mes .@new_slug_str$ + "数量：^3131FF" + .@new_count;
			next;
			mes "要兑换吗？";
			next;
			switch(select("兑换:取消")) {
				case 1:
					mes "[肯尼]";
					mes "完成了。";
					delitem .@slug_id, .@new_count * .@req_count;
					getitem 25187, .@new_count;
					close;
				case 2:
					close;
			}
		case 2:
			mes "[肯尼]";
			mes "完成了。";
			delitem .@slug_id, .@req_count;
			getitem 25187, 1;
			close;
		case 3:
			callsub S_Close;
	}

S_Close:
	mes "[肯尼]";
	mes "好的。";
	mes "如果还需要什么，请再回来。";
	close;
}
