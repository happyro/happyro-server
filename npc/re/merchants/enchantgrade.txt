grademk,34,184,4	script	Sratos#sratos	4_JP_GARM_H,{
	mes "[斯拉托斯]";
	mes "你好，亲爱的顾客。愿今天有奇迹发生。";
	mes "我能为你做什么？";
	next;
	switch( select( "装备升阶", "兑换埃特尔材料", "什么是装备升阶？" ) ){
		case 1:
			mes "[斯拉托斯]";
			mes "顾客，你想为装备升阶吗？";
			mes "转移宝石的魔力并不容易。";
			next;
			mes "[斯拉托斯]";
			mes "魔力可能当场爆炸。";
			mes "继承魔力的武器也会随之爆炸！";
			next;
			switch( select( "我还是要做！", "让我考虑一下。" ) ){
				case 1:
					mes "[斯拉托斯]";
					mes "祝勇敢的顾客好运！";
					close2;
					enchantgradeui();
					end;
				case 2:
					mes "[斯拉托斯]";
					mes "鼓起勇气挑战时，请再回来。";
					close;
			}
		case 2:
			mes "[斯拉托斯]";
			mes "这些是用于装备升阶的材料。";
			mes "以太星尘是基础材料，支付费用后可以兑换以太魔石，再用以太魔石和宝石兑换升阶所需的宝石。";
			close2;
			callshop "EnchantGradeExchange";
			end;

		case 3:
			mes "[斯拉托斯]";
			mes "有时会发现蕴含大量纯粹魔力的不稳定宝石。";
			mes "我们称它们为埃特尔宝石，对吧？";
			next;
			mes "[斯拉托斯]";
			mes "你可以把埃特尔宝石的魔力转移到其他武器上，但当然存在风险。";
			mes "过程很精细，一不小心，啪！就会爆炸。";
			next;
			mes "[斯拉托斯]";
			mes "总之，使用埃特尔宝石可以提升你所用武器的阶级。";
			mes "性能会变得更好。我们称之为装备升阶。";
			next;
			mes "[斯拉托斯]";
			mes "如果心爱的武器更上一层楼，你也会高兴吧？";
			mes "我们就是帮助它实现的人。虽然有点风险，啊哈哈哈。";
			close;
	}

OnInit:
	setunittitle(getnpcid(0), "<装备升阶师>");
	end;
}
