-	script	novice_vending_machine#main	-1,{
	.@storage_ticket = 7059;	// Cargo_Free_Ticket
	.@storage_ticket_name$ = getitemname(.@storage_ticket);

	mes "初心者自动售货机。";
	mes "这里有很多按钮。";
	next;
	switch( select( "说明", "兑换 " + .@storage_ticket_name$, "兑换新手装备" ) ) {
	case 1:
		mes "[兑换 " + .@storage_ticket_name$ + "]";
		mes "4 个" + .@storage_ticket_name$ + "可兑换 1 张免费传送、飞艇或手推车服务券，或 1 张免费大嘴鸟骑乘券、15 个新手苍蝇翅膀、5 个[非卖品]新手专用蝴蝶翅膀、30 个初心者药水。";
		next;
		mes "[兑换新手装备]";
		mes "初学者忍者斗篷、初学者阴险披肩、初学者海滩鞋、初心者左手短剑和初学者铁盾，每件可兑换 4 个" + .@storage_ticket_name$ + "。";
		next;
		mes "[兑换新手装备]";
		mes "每个初学者蛋壳帽可兑换一张克里图拉美发券，用于更换发型。";
		close;
	case 2:
		setarray .@exchange[0],
			7060,1,		// Warp_Free_Ticket
			7311,1,		// Free_Flying_Ship_Ticket
			7061,1,		// Cart_Free_Ticket
			7310,1,		// Free_Peco_Ticket
			12323,15,	// N_Fly_Wing
			12324,5,	// N_Butterfly_Wing
			569,30;		// Novice_Potion
		.@size = getarraysize(.@exchange);

		for ( .@i = 0; .@i < .@size; .@i += 2 ) {
			if (.@exchange[.@i+1] > 1)
				.@menu$ += .@exchange[.@i+1] + " ";
			.@menu$ += getitemname(.@exchange[.@i]) + ":";
		}
		.@s = (select(.@menu$) - 1) * 2;
		if (countitem(.@storage_ticket) < 4) {
			mes .@storage_ticket_name$ + "数量不足，无法进行兑换。";
			close;
		}
		if (!checkweight(.@exchange[.@s],.@exchange[.@s+1])) {
			mes "请先为兑换物品预留足够的负重和物品栏空间。";
			close;
		}
		if (.@exchange[.@s+1] > 1)
			.@amount$ = .@exchange[.@s+1] + " ";
		mes "已将 4 个" + .@storage_ticket_name$ + "兑换为" + .@amount$ + getitemname(.@exchange[.@s]);
		delitem .@storage_ticket,4;
		getitem .@exchange[.@s],.@exchange[.@s+1];
		close;
	case 3:
		mes "将所有新手装备兑换为优惠券。";
		mes "包含正在穿戴、已精炼或插卡的上述新手装备，请确认后再继续。";
		next;
		if (select( "继续", "取消。" ) == 2) {
			mes "交易已中断。";
			close;
		}
		mes "正在将所有可兑换的新手装备兑换为优惠券。";
		setarray .@list[0],2352,2510,2414,1243,2112,5055;
		.@size = getarraysize(.@list);
		for (.@i = 0; .@i < .@size; .@i++) {
			.@count = countitem(.@list[.@i]);
			if (.@list[.@i] == 5055)
				.@hair_coupons += .@count;
			else
				.@storage_tickets += 4 * .@count;
		}
		if (.@storage_tickets > 0) {
			.@reward_ids[.@reward_size] = .@storage_ticket;
			.@reward_amounts[.@reward_size++] = .@storage_tickets;
		}
		if (.@hair_coupons > 0) {
			.@reward_ids[.@reward_size] = 6593;
			.@reward_amounts[.@reward_size++] = .@hair_coupons;
		}
		if (.@reward_size > 0 && !checkweight2(.@reward_ids,.@reward_amounts)) {
			mes "请先为全部兑换券预留足够的负重和物品栏空间。";
			close;
		}
		
		for ( .@i = 0, .@i < .@size; .@i++ ) {
			.@count = countitem(.@list[.@i]);	// note: maybe check if equipped items are skipped
			if (.@count < 1)
				continue;
			switch(.@list[.@i]) {
			case 2352:	// Novice_Plate
			case 2510:	// Novice_Hood
			case 2414:	// Novice_Boots
			case 1243:	// Novice_Knife
			case 2112:	// Novice_Guard
				delitem .@list[.@i], .@count;
				getitem .@storage_ticket,(4*.@count);	// Cargo_Free_Ticket
				break;
			case 5055:	// Novice_Egg_Cap
				delitem .@list[.@i], .@count;
				getitem 6593,.@count;	// Cryptura_Hair_Coupon
				break;
			}
		}
		close;
	}
}
