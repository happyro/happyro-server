-	script	#GemstoneBagger	HIDDEN_NPC,{
	mes "[" + strnpcinfo(1) + "]";
	mes "我们会将宝石装入袋中，方便您携带。";
	mes "您想做什么？";
	next;
	switch(select("兑换宝石:取消")) {
	case 1:
		mes "[" + strnpcinfo(1) + "]";
		mes "我会准备一个宝石袋。";
		close2;
		callshop "Gemstone_Bagger";
		break;
	case 2:
		mes "[" + strnpcinfo(1) + "]";
		mes "需要帮助时再回来。";
		close;
	}
}
