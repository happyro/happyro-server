		if (rentalcountitem(getarg(.@i)) > isequippedcnt(getarg(.@i)))// don't count item equipped
			.@menu$ = .@menu$ + "^4d4dff"+ getarg(.@i+1) +" - 已借出^000000:";
		else
			.@menu$ = .@menu$ + getarg(.@i+1) + ":";
	}
	.@total_item = (getargcount() -1) /2;
	.@s = select( .@menu$ + ":查看其他物品" ) -1;
	if (.@s <= .@total_item) {
		.@index = .@s *2 +1;
		mes "[装备租赁管理员]";
		if (rentalcountitem(getarg(.@index)) > isequippedcnt(getarg(.@index)))
			mes "你已经租用了"+ getarg(.@index+1) +".";
		else {
			mes "请收下"+ getarg(.@index+1) +".";
			mes "^4d4dff这件装备的租用期限为 24 小时^000000，到期后会自动消失。";
			rentitem getarg(.@index),86400;
		}
		mes "还需要其他装备吗？";
		next;
	}
	return;
