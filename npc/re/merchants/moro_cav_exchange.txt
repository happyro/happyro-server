moro_cav,32,73,5	script	Hero Token Exchanger#1	4_F_SHADOWCHASER,{
	disable_items;
	mes "[Hero Token Exchanger]";
	mes "如果您与梦罗克战斗过并收集了英雄徽章，我可以为您兑换一枚特殊戒指。";
	mes "想看看我的戒指吗？";
	while(true) {
		next;
		if (checkweight(2163,2) == 0) {
			mes "[Hero Token Exchanger]";
			mes "您的背包里有太多物品，整理后再回来吧！";
			close;
		}
		switch( select( "戒指说明", "兑换戒指。", "结束对话。" ) ) {
		case 1:
			mes "[Hero Token Exchanger]";
			mes "这枚英雄戒指看起来也许并不起眼。";
			next;
			mes "[Hero Token Exchanger]";
			mes "但把它带给附魔专家 伯恩斯，他可以为 4 个插槽附魔，唤醒它潜藏的能力。";
			mes "";
			next;
			mes "[Hero Token Exchanger]";
			mes "英雄戒指可以帮助您征讨梦罗克。";
			mes "还有其他问题吗？";
			continue;
		case 2:
			mes "- 英雄戒指 -";
			mes "^006400为征讨梦罗克的战士打造的戒指。经过附魔后，它将赋予佩戴者对抗梦罗克的力量。^000000";
			mes "^006400类型：饰品 / 防御：0^000000";
			mes "^006400重量：20 / 需求等级：160^000000";
			mes "^006400插槽：0 / 适用职业：所有职业^000000";
			next;
			mes "[Hero Token Exchanger]";
			mes "兑换英雄戒指需要 30 个英雄徽章。";
			mes "要兑换吗？";
			next;
			if (select( "返回上级菜单。", "兑换 30 个英雄徽章。" ) == 1) {
			mes "[Hero Token Exchanger]";
				mes "请再考虑一下。";
				continue;
				}
			if (countitem(6684) < 30) {
			mes "[Hero Token Exchanger]";
				mes "^006400英雄徽章^000000";
				mes "英雄徽章会赠予勇敢对抗梦罗克的人。";
				mes "您的英雄徽章不足。需要 ^00640030 个英雄徽章^000000才能兑换。";
				continue;
				}
			mes "[Hero Token Exchanger]";
			mes "我会用 ^00640030 个英雄徽章^000000为您兑换 1 枚 ^006400英雄戒指^000000。";
			mes "还需要别的吗？";
			delitem 6684,30;	// TokenOfHero
			getitem 2981,1;		// RingOfHero
			continue;
			case 3:
		mes "[Hero Token Exchanger]";
			mes "总有一天您会发现自己需要英雄戒指。";
			close;
			}
		}
	}

moro_cav,34,65,5	script	Expert Enchanter Byrnes	1_M_03,{
	disable_items;
	mes "[伯恩斯]";
	mes "您听说过英雄戒指吗？只有真正的英雄才配拥有那枚华丽的戒指，我一直在等待这样的英雄。";
	mes "我能为您做些什么？";
	next;
	switch( select( "英雄戒指附魔。", "重置英雄戒指附魔。", "取消。" ) ) {
	case 1:
		mes "[伯恩斯]";
		mes "我只能为已装备的戒指附魔。您想为哪一枚已装备的戒指附魔？";
		next;
		setarray .@eqi_slot[0], EQI_ACC_L, EQI_ACC_R;
		.@s = select( "左侧饰品", "右侧饰品" ) - 1;
		switch( getequipid(.@eqi_slot[.@s]) ) {
		case -1:
			mes "[伯恩斯]";
			mes "请装备想要附魔的英雄戒指。";
			close;
		case 2981:
			break;
		default:
			mes "[伯恩斯]";
			mes "请装备英雄戒指。我无法为其他物品附魔。";
			close;
		}
		// custom at this point
		setarray .@card[0],
			getequipcardid(.@eqi_slot[.@s],0),
			getequipcardid(.@eqi_slot[.@s],1),
			getequipcardid(.@eqi_slot[.@s],2),
			getequipcardid(.@eqi_slot[.@s],3);

		if (.@card[0] > 0) {
			mes "[伯恩斯]";
			mes "已经完成全部附魔的英雄戒指无法继续附魔。";
			close;
		}
		if (.@card[3] == 0) {
			setarray .@enchant_list$[0], "DEF", "MDEF";
			.@slot = 4;
		}
		else if (.@card[2] == 0) {
			setarray .@enchant_list$[0], "Max HP", "Max SP";
			.@slot = 3;
		}
		else if (.@card[1] == 0) {
			setarray .@enchant_list$[0], "Atk", "MAtk";
			.@slot = 2;
		}
		else {
			setarray .@enchant_list$[0], "STR", "INT", "AGI", "VIT", "DEX", "LUK";
			.@slot = 1;
		}
		mes "[伯恩斯]";
		mes "您想在插槽 " + .@slot + " 上附魔哪种能力？";
		next;
		.@enchant_list_size = getarraysize(.@enchant_list$);
		for ( .@i = 0; .@i < .@enchant_list_size; ++.@i )
			.@menu$ += .@enchant_list$[.@i] + ":";
		.@type = select(.@menu$) - 1;

		switch(.@slot) {
		case 4:
			switch(.@type) {// enchant list index
			case 0:	// DEF
				setarray .@bonus[0], 4791, 4792, 4793, 4794, 4893;
				break;
			default:	// MDEF
				setarray .@bonus[0], 4890, 4786, 4891, 4787, 4892;
				break;
			}
			break;
		case 3:
			switch(.@type) {
			case 0:	// Max HP
				setarray .@bonus[0], 4861, 4862, 4867, 4868, 4900;
				break;
			default:	// Max SP
				setarray .@bonus[0], 4800, 4870, 4871, 4801, 4802;
				break;
			}
			break;
		case 2:
			switch(.@type) {
			case 0:	// Atk
				setarray .@bonus[0], 4882, 4766, 4767, 4894, 4895;
				break;
			default:	// Matk
				setarray .@bonus[0], 4883, 4896, 4897, 4898, 4899;
				break;
			}
			break;
		case 1:
			switch(.@type) {
			case 0:	// STR
				setarray .@bonus[0], 4700, 4701, 4702, 4703, 4704;
				break;
			case 1:	// INT
				setarray .@bonus[0], 4710, 4711, 4712, 4713, 4714; 
				break;
			case 2:	// AGI
				setarray .@bonus[0], 4730, 4731, 4732, 4733, 4734;
				break;
			case 3:	// VIT
				setarray .@bonus[0], 4740, 4741, 4742, 4743, 4744;
				break;
			case 4:	// DEX
				setarray .@bonus[0], 4720, 4721, 4722, 4723, 4724;
				break;
			default:	// LUK
				setarray .@bonus[0], 4750, 4751, 4752, 4753, 4754;
				break;
			}
			break;
		}

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@eqi_slot[.@s], 2981) || callfunc("F_IsEquipCardHack", .@eqi_slot[.@s], .@card[0], .@card[1], .@card[2], .@card[3]))
			close;

		.@bonus_size = getarraysize(.@bonus);
		.@card[.@slot-1] = .@bonus[ rand(.@bonus_size) ];	// unknown rates
		specialeffect2 EF_ANGEL2;
		mes "[伯恩斯]";
		mes "附魔完成！";
		delequip .@eqi_slot[.@s];
		getitem2 2981,1,1,0,0,.@card[0],.@card[1],.@card[2],.@card[3];
		close;
	case 2:
		mes "[伯恩斯]";
		mes "要重置英雄戒指的附魔，我需要 3 个英雄徽章。";
		next;
		if (countitem(6684) < 3) {
			mes "[伯恩斯]";
			mes "您的英雄徽章不足，无法重置英雄戒指。";
			close;
		}
		mes "[伯恩斯]";
		mes "我只能重置已装备的戒指。您想重置哪一枚？";
		next;
		setarray .@eqi_slot[0], EQI_ACC_L, EQI_ACC_R;
		.@s = select( "左侧饰品", "右侧饰品" ) - 1;
		switch( getequipid(.@eqi_slot[.@s]) ) {
		case -1:
			mes "[伯恩斯]";
			mes "请先装备要重置附魔状态的物品。";
			close;
		case 2981:
			break;
		default:
			mes "[伯恩斯]";
			mes "请装备英雄戒指。我无法重置其他物品。";
			close;
		}
		// custom at this point
		if (countitem(6684) < 3) {
			mes "[伯恩斯]";
			mes "您的英雄徽章不足，无法重置英雄戒指。";
			close;
		}
		setarray .@card[0],
			getequipcardid(.@eqi_slot[.@s],0),
			getequipcardid(.@eqi_slot[.@s],1),
			getequipcardid(.@eqi_slot[.@s],2),
			getequipcardid(.@eqi_slot[.@s],3);

		if (.@card[3] == 0) {
			mes "[伯恩斯]";
			mes "您的英雄戒指当前没有附魔。";
			close;
		}

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@eqi_slot[.@s], 2981) || callfunc("F_IsEquipCardHack", .@eqi_slot[.@s], .@card[0], .@card[1], .@card[2], .@card[3]))
			close;

		specialeffect2 EF_ANGEL2;
		mes "[伯恩斯]";
		mes "初始化完成。";
		delitem 6684,3;
		delequip .@eqi_slot[.@s];
		getitem 2981,1;
		close;
	case 3:
		close;
	}
}
