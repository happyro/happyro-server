izlude,127,175,3	script	Hypnotist#novice	124,{
	mes "[催眠师]";
	mes "你好，冒险者。";
	mes "我是催眠师学院的成员，被派到依斯鲁得，为特定一转职业角色提供属性和技能重置服务。";
	mes "价格非常优惠：免费！";
	next;
	mes "[催眠师]";
	mes "目前提供无限次属性/技能重置，但需要符合职业与等级要求。";
	mes "跆拳少年、跆拳少女、神枪手和忍者的等级必须低于^FF0000基础等级 50^000000。";
	next;
	mes "[催眠师]";
	mes "可以使用服务的职业包括^FF0000剑士、服事、法师、盗贼、弓箭手、商人、跆拳少年或跆拳少女、神枪手和忍者^000000。";
	mes "现在，你有什么问题吗？";
	next;
	switch(select("^FF0000属性/技能重置？:重置属性:重置技能^000000:取消")) {
	case 1:
		mes "[催眠师]";
		mes "如果对当前属性或技能不满意，重置后就能重新分配相应点数。";
		next;
		mes "[催眠师]";
		mes "进行属性/技能重置前，必须将角色携带物品的总重量降为 0。可以先把物品放进卡普拉仓库。";
		next;
		mes "[催眠师]";
		mes "如果装备了^FF0000手推车^000000，也要先移除。";
		mes "否则催眠不会生效，甚至会反噬……";
		close;
	case 2:
		set .@str$,"属性";
	case 3:
		if (.@str$ == "")
			set .@str$,"技能";
		mes "[催眠师]";
		mes "确定要使用我的 ^FF0000"+.@str$+"重置^000000 服务吗？";
		next;
		if(select("是:取消") == 1)
			break;
	case 4:
		mes "[催眠师]";
		mes "谢谢，祝你冒险顺利。";
		mes "旅途平安~";
		close;
	}
	if ((Class >= Job_Swordman && Class <= Job_Thief) || Class == Job_Taekwon || Class == Job_Gunslinger || Class == Job_Ninja) {
		if (checkcart()) {
			mes "[催眠师]";
			mes "哦！请先移除手推车，再继续使用"+.@str$+"重置服务。";
			mes "感谢配合~";
			close;
		}
		if (Weight != 0) {
			mes "[催眠师]";
			mes "使用我的"+.@str$+"重置服务前，必须将^FF0000物品栏重量降为 0^000000。";
			next;
			mes "[催眠师]";
			mes "不如先把东西放进卡普拉仓库？这样就能安全保管所有物品。";
			mes "";
			close;
		}
		if (BaseLevel >= 50 && (Class == Job_Taekwon || Class == Job_Gunslinger || Class == Job_Ninja)) {
			mes "[催眠师]";
			mes "抱歉，跆拳少年、跆拳少女、神枪手和忍者达到基础等级 50 后，就不能使用我提供的"+.@str$+"重置服务了。";
			close;
		}
		mes "[催眠师]";
		mes "感谢使用我的"+.@str$+"重新分配服务。祝你冒险顺利、旅途愉快，冒险者。";
		mes "";
		if (.@str$ == "属性")
			resetstatus;
		else
			resetskill;
		close;
	} else {
		mes "[催眠师]";
		mes "抱歉，你的职业不符合我提供的"+.@str$+"重置服务要求。";
		mes "我只能为以下职业提供"+.@str$+"重置……";
		mes "";
		mes "";
		next;
		mes "[催眠师]";
		mes "^FF0000剑士、服事、法师、盗贼、弓箭手、商人、跆拳少年、跆拳少女、神枪手和忍者^000000。";
		close;
	}
}
