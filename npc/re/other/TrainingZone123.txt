tra_fild01,204,110,2	script	训练场仓库士兵#fild01_1	8W_SOLDIER,{
	mes "[训练场仓库士兵]";
	mes "要使用仓库吗？训练场仓库需要1000 Zeny。请选择要使用的仓库。";
	next;
	// switch( select( "¢º Storage 1", "¢º Storage 2", "¢º Storage 3", "¢º Guild Storage", "¢º Cancel" ) ) {
	switch( select( "仓库1", "", "", "公会仓库", "取消" ) ) {
	case 1:
		.@storage_num = 1;
		break;
	case 2:
		.@storage_num = 2;
		break;
	case 3:
		.@storage_num = 3;
		break;
	case 4:
		// copied from npc/re/merchants/guild_warehouse.txt
		.@guildid = getcharid( 2 );

		if( .@guildid == 0 ){
			mes "[训练场仓库士兵]";
			mes "公会仓库仅供公会成员使用。";
			close;
		}

		if( getgdskilllv( .@guildid, "GD_GUILD_STORAGE" ) == 0 || !guild_has_permission( GUILD_PERM_STORAGE ) ){
			mes "[训练场仓库士兵]";
			mes "你的公会似乎还不能使用公会仓库，或者你没有仓库权限。";
			mes "请确认公会技能和仓库权限后再回来。";
			close;
		}

		if( Zeny < 1000 ){
			mes "[训练场仓库士兵]";
			mes "恐怕你的钱不够支付费用。";
			mes "费用是1000 Zeny。";
			close;
		}

		mes "[训练场仓库士兵]";
		mes "那我为你打开公会仓库。祝你留下难忘回忆！";
		close2;

		if( Zeny < 1000 ){
			// Cheat prevention
			end;
		}

		if( guildopenstorage() == GSTORAGE_OPEN ){
			Zeny -= 1000;
			end;
		}else{
			mes "[训练场仓库士兵]";
			mes "恐怕现在有人正在使用仓库。";
			mes "过一会儿再试。";
			close;
		}
	case 5:
		end;
	}
	// TODO (custom text)
	if (!callfunc("F_CanOpenStorage")) {
		mes "[训练场仓库士兵]";
		mes "抱歉，你需要将初心者的基本技能提升到6级，才能使用仓库服务。";
		close2;
		cutin "",255;
		end;
	}

	if (Zeny < 1000) {
		mes "[训练场仓库士兵]";
		mes "^4d4dff使用训练场仓库需要1000 Zeny。^000000";
		close;
	}
	Zeny -= 1000;
	openstorage; // TODO: open correct storage 1-3
	end;
}

tra_fild01,142,113,6	script	清扫机器人#fild01	EP17_2_OMEGA_CLEANER,{
OnStart:
	.@num = atoi( charat(strnpcinfo(2), 5) );

	if (.is_moving[.@num])
		end;
	.is_moving[.@num] = true;
	stopnpctimer;

	setarray .@emotions[0], ET_SWEAT, ET_PROFUSELY_SWEAT, ET_HUNGRY, ET_OHNO;
	.@r = rand( getarraysize(.@emotions) );
	emotion .@emotions[.@r];

	setarray .@speed[0], 300, 400, 500, 600;
	.@r = rand( getarraysize(.@speed) );
	npcspeed .@speed[.@r];

	getmapxy .@map$,.@x,.@y, BL_NPC;
	switch( .@num ) {
	case 1:
		if (.@x <= 126)
			.sens[.@num] = 1;
		else if (.@x >= 198)
			.sens[.@num] = -1;
		.@x += 8 * .sens[.@num];
		break;
	case 2:
		if (.@x <= 21)
			.sens[.@num] = 1;
		else if (.@x >= 85)
			.sens[.@num] = -1;
		.@x += 8 * .sens[.@num];
		break;
	case 3:
		if (.@y <= 33)
			.sens[.@num] = 1;
		else if (.@y >= 89)
			.sens[.@num] = -1;
		.@y += 8 * .sens[.@num];
		break;
	}
	unitwalk getnpcid(0),.@x,.@y, strnpcinfo(0) + "::OnMoveEnd";
	end;
OnMoveEnd:
	.@num = atoi( charat(strnpcinfo(2), 5) );
	.is_moving[.@num] = false;
	initnpctimer;
	end;
OnTimer30000:
	donpcevent strnpcinfo(0) + "::OnStart";
	end;

OnInit:
	.@num = atoi( charat(strnpcinfo(2), 5) );
	.sens[.@num] = 1;
	initnpctimer;
	end;
}
