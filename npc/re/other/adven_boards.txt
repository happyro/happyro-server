function	script	F_AvenBoard	{
	.@mo_board = select( "按楼层查找怪物",
					"怪物信息检索系统",
					( countitem(909) == 30000 ? "请求" : "" ),
					"取消" );
	if (.@mo_board == 1)
		return 1;
	else if (.@mo_board == 2) {
		while(1) {
			mes "[怪物信息检索系统]";
			mes "这里是怪物信息检索系统。可以搜索出现在该地下城中的怪物信息。搜索时请输入怪物名称，包括空格。";
			next;
			input .@inputstr$;
			setarray .@element$[0],"无属性","水","地","火","风","毒","圣","暗","念","不死";
			setarray .@size$[0],"小型","中型","大型";
			setarray .@race$[0],"无形","不死","动物","植物","昆虫","鱼类","恶魔","人形","天使","龙族";
			.@size = getargcount();
			.@i = 0;
			while( .@i < .@size && getarg(.@i+1,"") != .@inputstr$ ) .@i += 3;
			mes "[怪物信息检索系统]";
			mes "正在搜索";
			next;
			mes "[怪物信息检索系统]";
			mes "搜索完成。";
			mes "现在显示搜索结果。";
			next;
			mes "[怪物信息检索系统]";
			if (.@i >= .@size) {
				mes "没有该怪物的信息。";
				mes "请检查怪物名称。";
			}
			else {
				.@mob_id = getarg(.@i);
				mes "- 名称："+ getarg(.@i+1);
				mes "- 属性："+ .@element$[ getmonsterinfo(.@mob_id,MOB_ELEMENT) ];
				mes "- 大小："+ .@size$[ getmonsterinfo(.@mob_id,MOB_SIZE) ];
				mes "- 种族："+ .@race$[ getmonsterinfo(.@mob_id,MOB_RACE) ];
				mes "--------------------------------^FD8D02";
				mes getarg(.@i+2);
			}
			next;
			mes "[怪物信息检索系统]";
			mes "要搜索其他怪物信息吗？";
			next;
			if (select( "搜索","取消" ) == 1)
				continue;
			mes "[怪物信息检索系统]";
			mes "怪物搜索系统现在关闭。";
			next;
			return 2;
		}
	}
	else if (.@mo_board == 3) {
		mes "请求的信息会显示在这里——显示出来就没有问题。";
		next;
		return 3;
	}
	else if (.@mo_board == 4)
		close;
	else {
		mes "操作错误。";
		close;
	}
}

function	script	F_InfoBoard	{
	next;
	mes "[冒险者公会]";
	mes "如需了解更多怪物信息，请使用怪物信息检索系统。";
	next;
	return;
}
