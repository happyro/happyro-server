// Channel Warper :: channel
//============================================================
function	script	F_IzludeChannel	{
	mes "[频道传送员]";
	if (!getbattleflag("navigation_map_channels_enabled")) {
		mes "当前仅开放频道 1。";
		close;
	}
	mes "依斯鲁得是个非常热闹的地方，对吧？";
	mes "我来帮助依斯鲁得变得更顺畅！";
	mes "我可以把你送到另一个依斯鲁得分区。";
	mes "要前往另一个分区吗？";
	next;
	set .@i, select(getarg(0)+":算了");
	if (.@i < 6) {
		setarray .@maps$[1],"izlude","izlude_a","izlude_b","izlude_c","izlude_d";
		savepoint .@maps$[.@i],128,98,1,1;
		warp .@maps$[.@i],128,98;
		end;
	}
	close;
}
