				erasequest .@main_quest;
				close;
			}
			if (.@time_check != 1 && .@siege_check != 1) {
				mes "你目前没有接受这项任务。";
				mes "请再确认一下。";
				close;
			}
			if (.@mission02 == 1 || .@mission03 == 1) {
				mes "你已经接受了内容相同的任务，";
				mes "虽然难度不同。";
				mes "除非你打算取消当前任务，否则请先把它完成。";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "这样才合理，不是吗？";
				mes "你不这么认为吗？";
				close;
			}
			mes "嗯，你可能不相信，这项任务与一宗盗窃案有关。";
			mes "他们说有一批"+ .@string$[2] +"从以下地区的要塞里失窃了："+ .@fortress$ +".";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "小偷大概趁攻城战混乱时下手了。";
			mes "这项委托需要找回至少 ^4d4dff"+ .@num_items +"个"+ .@string$[2] +"。^000000";
			callsub( S_Acceptquest, .@main_quest, .@npc_name$ );
		}
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "TE 区域的气氛比较轻松，";
		mes "光打攻城战也不够有趣，对吧？";
		mes "所以，我准备了^4d4dff一些可以在攻城战期间完成的工作^000000～";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "你可以在^4d4dffTE 攻城战开始前一小时至结束期间^000000接受任务，";
		mes "每项任务都必须在 ^4d4dff4 小时内^000000完成。";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "简单来说，就是在规定的攻城时段向我接受任务，";
		mes "然后^4d4dff在时限内回来向我报告^000000。";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "就当来玩玩吧。";
		close;
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "我明白了。";
		mes "改变主意的话，随时回来。";
		mes "不过，我可不一定有空。嘿嘿。";
		close;
	}

S_Cancelquest:
	.@npc_name$ = getarg(1);
	if (WoeTETimeStart(3600) == 1 || agitcheck3()) {
		mes "看来你还没完成任务？";
		mes getarg(2);
		next;
		mes "["+ .@npc_name$ +"]";
		mes "你想取消任务吗？";
		next;
		switch( select( "不。","是。" ) ) {
		case 1:
			mes "["+ .@npc_name$ +"]";
			mes "祝你好运！";
			close;
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "按你的要求，这项任务已取消。";
			erasequest getarg(0);
			close;
		}
	}
	else {
		mes "嗯，你还剩一些时间，但现在还不是攻城时间。";
		mes "我觉得你不该继续。";
		mes "你应该取消这项任务，我认为它已经失败了。";
		next;
		switch( select( "我想继续。","好的，我要取消。" ) ) {
		case 1:
			mes "["+ .@npc_name$ +"]";
			mes "我觉得不值得。好吧，祝你好运！";
			close;
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "按你的要求，这项任务已取消。";
			erasequest getarg(0);
			close;
		}
	}

S_Acceptquest:
	.@npc_name$ = getarg(1);
	next;
	switch( select( "接受。","不接受。" ) ) {
	case 1:
		mes "["+ .@npc_name$ +"]";
		mes "^4d4dff时间限制为 4 小时。^000000";
		mes "请在规定时间内完成任务并向我报告。";
		setquest getarg(0);
		close;
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "那就试试其他任务吧。";
		close;
	}

S_Rentbox:
	setarray .@item_id[0], 2950,15067,20709,22012;// Rune_Ring, Rune_Suit, Mana_Manteau, Mana_Boots
	for ( .@count = 0; .@count < getarg(0); .@count++ ) {
		.@miss_item01 = rand(100);
		if (.@miss_item01 > 3)
			mes "你什么也没找到。";
		else {
			rentitem .@item_id[ .@miss_item01 ],86400;
			mes "你找到了‘"+ getitemname(.@item_id[ .@miss_item01 ]) +"！‘";
		}
	}
	close;

S_Menu:
	for ( .@i = 1; .@i < getargcount(); .@i += 2 ) {
		if ( isbegin_quest(getarg(.@i)) )
			.@menu$ = .@menu$ + getarg(.@i+1) + " ^4d4dff查看结果^000000:";
		else
			.@menu$ = .@menu$ + getarg(.@i+1) + ":";
		set getelementofarray( getarg(0),.@index ), getarg(.@i);
		.@index++;
	}
	return .@menu$;
}
