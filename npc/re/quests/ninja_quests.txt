que_ng,21,72,6	script	防具制作大师	4_M_RASWORD,{
	mes "[行]";
	mes "你如何定义胜利？";
	mes "击倒敌人……";
	mes "然而，最重要的是……";
	mes "永远不要让自己倒下！";
	next;
	mes "[行]";
	mes "为此，你必须更好地保护自己。";
	mes "如果连自己都保护不好，就无法保护‘他人’。";
	next;
	if(select("合成护甲。:结束对话。") == 2) {
		mes "[行]";
		mes "谨慎也是战斗中的重要美德。";
		close;
	}
	switch(select("狼纹护手[1]:月纹护手[1]:忍者鳞甲[1]:雾中暗影[1]:退出。")) {
	case 1:
		mes "[行]";
		mes "狼纹护手……一件野兽般的护甲。";
		next;
		callsub L_Select,2172;
		if (countitem(2171) > 0 && countitem(6252) > 9) {  //custom translation
			delitem 2171,1; //Fox_Armguard
			delitem 6252,10; //Wolf_Blood
			getitem 2172,1; //Wolf_Armguard
			mes "[行]";
			mes "你感受到狼的本性了吗？";
			mes "既野性又迷人……";
			close;
		}
		break;
	case 2:
		mes "[行]";
		mes "月纹护手……美丽与力量的结合……两者缺一不可。";
		next;
		callsub L_Select,2173;
		if (countitem(2171) > 0 && countitem(7321) > 99) {  //custom translation
			delitem 2171,1; //Fox_Armguard
			delitem 7321,100; //Fragment_Of_Crystal
			getitem 2173,1; //Crescent_Armguard
			mes "[行]";
			mes "真美……";
			mes "仿佛全身都溢满了魔力。";
			close;
		}
		break;
	case 3:
		mes "[行]";
		mes "忍者鳞甲……据说曾有传奇忍者穿过它。";
		next;
		callsub L_Select,15054;
		if (countitem(7451) > 29 && countitem(7562) > 29 && countitem(6091) > 29) {  //custom translation
			delitem 7451,30; //Scale_Of_Red_Dragon
			delitem 7562,30; //Ice_Scale
			delitem 6091,30; //Dark_Red_Scale
			getitem 15054,1; //Ninja_Scale_Armor
			mes "[行]";
			mes "这确实是传奇护甲……";
			mes "即使我并不相信传说……";
			close;
		}
		break;
	case 4:
		mes "[行]";
		mes "雾中暗影……它是为了保护他人而制作的。";
		next;
		callsub L_Select,15055;
		if (countitem(15056) > 0 && countitem(6089) > 9) {  //custom translation
			delitem 15056,1; //Special_Ninja_Suit_
			delitem 6089,10; //Piece_Of_Darkness
			getitem 15055,1; //Tenebris_Latitantes
			mes "[行]";
			mes "怎么样？";
			mes "这件护甲看起来不像有两件吗？";
			close;
		}
		break;
	case 5:
		close;
	}
	mes "[行]";
	mes "你的材料不够。";
	mes "再检查一下……";
	close;

L_Select:
	.@string1$ = "检查材料。";
	.@string2$ = "检查属性。";
	while(1) {
		switch(select(.@string1$,.@string2$,"合成。","退出。")) {
		case 1:
			.@string1$ = "";
			callsub L_Ingredients, getarg(0);
			break;
		case 2:
			.@string2$ = "";
			callsub L_Stats, getarg(0);
			break;
		case 3:
			callsub L_Combine;
			return;
		case 4:
			mes "[行]";
			mes "看来你需要更多时间考虑。";
			close;
		}
	}

L_Ingredients:
	mes "[行]";
	switch (getarg(0)) {
	case 2172:	//Wolf_Armguard
		mes "制作^ff0000狼纹护手需要 1 个狐纹护手和 10 份狼血^000000。";
		mes "你能感受到吗？狼血正在跳动……";
		break;
	case 2173:	//Crescent_Armguard
		mes "制作^ff0000月纹护手需要 1 个狐纹护手和 100 个水晶碎片^000000。";
		mes "上面会刻有如水晶般美丽清澈的新月。";
		break;
	case 15054:	//Ninja_Scale_Armor
		mes "制作^ff0000忍者鳞甲需要 30 个脱落的红色龙鳞、30 个冰鳞和 30 个红鳞碎片^000000。";
		mes "你能感受到它的传奇力量。";
		break;
	case 15055:	//Tenebris_Latitantes
		mes "制作^ff0000雾中暗影需要带插槽的特攻忍装和 10 个黑暗碎片^000000。";
		mes "黑暗会抹去存在本身……";
		break;
	}
	next;
	return;

L_Stats:
	mes "[" + getitemname(getarg(0)) + "]";
	switch (getarg(0)) {
	case 2172:	//Wolf_Armguard
		mes "近战攻击时有一定概率获得 5 秒 ATK+100、FLEE -50。";
		mes "盾牌类型，防御 70，插槽 1";
		mes "要求等级 100，仅限忍者类型";
		break;
	case 2173:	//Crescent_Armguard
		mes "每次精炼使技能后延迟减少 2%。";
		mes "盾牌类型，防御 45，插槽 1";
		mes "要求等级 100，仅限忍者类型";
		break;
	case 15054:	//Ninja_Scale_Armor
		mes "MHP+15%，MSP-30%";
		mes "护甲类型，防御 90，插槽 1";
		mes "要求等级 100，仅限忍者类型";
		break;
	case 15055:	//Tenebris_Latitantes
		mes "受到近战攻击时，有概率发动 1 级幻影—暗影。";
		mes "护甲类型，防御 60，插槽 1";
		mes "要求等级 100，仅限忍者类型";
		break;
	}
	next;
	return;

L_Combine:
	mes "[行]";
	mes "请确认^ff0000手边有正确的材料和装备^000000。";
	mes "事后后悔也没有用。";
	next;
	switch(select("合成。:退出。")) {
	case 1:
		if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
			mes "- 等一下！！ -";
			mes "- 你无法领取物品 -";
			mes "- 因为携带的物品太多。 -";
			mes "- 请减轻负重后 -";
			mes "- 再试一次。 -";
			close;
		}
		return;
	case 2:
		mes "[行]";
		mes "看来你需要更多时间考虑。";
		close;
	}
	end;
}
