		mes "坚实的灰色手推车。";

		switch( select( "什么是装饰手推车？", "学习装饰手推车", "结束对话" ) ) {

			mes "[杰森]";
			mes "我想彻底改造这辆可怜的手推车。";
			next;
			mes "[杰森]";
			mes "小小的改造还不够，你同意吗？";

			next;

				mes "学习装饰手推车很困难。";

			mes "[杰森]";
			mes "你带来了可爱手推车改造券。确定要学习装饰手推车吗？";

			if (select("是", "否") == 2) {
				mes "[杰森]";
				mes "改变主意后再来！";
				close;
			}
			mes "[杰森]";

			mes "然后加上这个……";
			mes "……";

			mes "[杰森]";

				mes "学习需要商人系职业和 1 张可爱手推车改造券，请确认后再来。";
				close;
			}
			delitem 6606,1; // P_Cart_C

			skill "MC_CARTDECORATE",1,SKILL_PERM;

			mes "现在轮到你试试了。";
			mes "装备手推车后，打开技能窗口，使用装饰手推车选择新外观。";

		mes "哦？让我看看。";

			mes "会使用手推车后再来吧！";
			close;
		}

		mes "你不记得我教过你什么了吗？";

		mes "[杰森]";
		mes "忘记了吗？";
		mes "现在来回忆一下！";
		next;
		mes "[杰森]";
		if (BaseClass != Job_Merchant) {
			mes "会使用手推车后再来吧！";
			close;
		}
		mes "仔细想想，你会再次记起来的。";
