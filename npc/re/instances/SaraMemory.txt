dali,139,118,4	script	次元移动机#sara::次元装置#sara	10007,{
	if (BaseLevel < 99) {
		mes "[冒险者利昂]";
		mes "嘿！你太年轻，不能玩那个！达到^0000FF 99 级^000000后再回来！";
		close;
	}
	if ((saramemory < 1) || (saramemory == 2) || (saramemory == 3) || (saramemory == 5)) {
		mes "[冒险者利昂]";
		mes "在你摆弄那个之前，过来和我聊聊吧？";
		close;
	}

	// 20 hour cooldown
	set .@ins_sara_time,checkquest(15002,PLAYTIME);
	if (.@ins_sara_time == -1) {

		set .@party_id,getcharid(1);
		set .@p_name$,getpartyname(.@party_id);
		set .@md_name$,"莎拉的记忆";

		if (!.@party_id) {
			mes "[冒险者利昂]";
			mes "进入前需要组建队伍！不知道怎么做吗？输入 /organize 队伍名称。";
			close;
		}

		if (is_party_leader() == true)
			set .@menu$, "启动次元装置。:使用次元装置。:取消。";
		else
			set .@menu$, ":使用次元装置。:取消。";

		switch(select(.@menu$)) {
		case 1:
			if (instance_create(.@md_name$) < 0) {
				mes "队伍名称："+.@p_name$;
				mes "队长："+strcharinfo(0);
				mes "^0000ff"+.@md_name$+" ^000000- 预约失败！";
				close;
			}
		mes "^FF0000次元装置顺利启动。使用装置进入时空裂缝。^000000";
			close;
		case 2:
			switch(instance_enter(.@md_name$)) {
			case IE_OTHER:
				mes "发生未知错误。";
				close;
			case IE_NOINSTANCE:
				mes "纪念副本"+.@md_name$+"不存在。";
				mes "队长还没有生成副本。";
				close;
			case IE_NOMEMBER:
				mes "只有登记成员才能进入副本"+.@md_name$+"。";
				close;
			case IE_OK:
				mapannounce "dali",strcharinfo(0)+"，队伍「"+.@p_name$+"」的成员，进入了副本「"+.@md_name$+"」。",bc_map,"0x00ff99";
				setquest 15002;
				end;
			}
		case 3:
		mes "^FF0000队长没有启动次元装置。^000000";
			close;
		}
	} else if (.@ins_sara_time == 0 || .@ins_sara_time == 1) {
		mes "[冒险者利昂]";
		mes "我想你上次进入裂缝后约 20 小时，它就会稳定到足以再次穿越。";
		close;
	}
	else {
		erasequest 15002;
		mes "^0000FF[莎拉的记忆] 冷却时间已结束。现在可以再次使用次元装置。^000000";
		close;
	}
}
