iz_dun04,130,234,5	script	入场守门员#iz_dun::Gatekeeper#iz_dun	413,{
	mes "[" + strnpcinfo(1) + "]";
	mes "站住！";
	mes "冒险者，你来这里有什么事？";
	mes "我们发现了这座海底城市的新一层。";
	next;
	mes "[" + strnpcinfo(1) + "]";
	mes "你准备好迎接挑战了吗？";
	mes "这里的魔物等级在 110 到 124 之间。";
	mes "还要当心魔鬼大乌贼！";
	next;
	switch(select("放出魔鬼大乌贼吧！:我还是留在这里。")) {
	case 1:
		mes "[" + strnpcinfo(1) + "]";
		mes "小心行事，祝你好运。";
		mes "如果遇到魔鬼大乌贼，一定要找同伴帮忙；不然就像我们一样，赶紧逃跑。";
		close2;
		warp "iz_dun05",141,187;
		end;
	case 2:
		mes "[" + strnpcinfo(1) + "]";
		mes "探索新区域之前，最好先做好准备。";
		mes "准备好出发后再来吧。";
		close;
	}
}
