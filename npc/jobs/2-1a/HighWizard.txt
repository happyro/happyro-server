valkyrie,44,47,5	script	High Wizard#Valkyrie	735,{
	if (ADVJOB == 0 ||  Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[进阶巫师]";
			mes "恭喜。";
			mes "向勇士致敬！";
			close;
		}
		mes "[进阶巫师]";
		mes "我们进阶巫师肩负着责任，使用毁灭性魔法实现正当目的。";
		next;
		mes "[进阶巫师]";
		mes "成为进阶巫师前需要一生的训练。";
		mes "你能想象吗，如果我们的力量落入坏人手中会发生什么？！";
		close;
	}
	else if (ADVJOB == Job_High_Wizard && Class == Job_Mage_High && JobLevel > 39) {
		mes "[进阶巫师]";
		mes "是时候了。";
		mes "米德加尔特需要能够驾驭最强魔法的人……";
		next;
		mes "[进阶巫师]";
		mes "你想成为进阶巫师吗？";
		next;
		if (select("不。:是。") == 1) {
			mes "[进阶巫师]";
			mes "准备好后，随时回来。";
			mes "向勇士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[进阶巫师]";
			mes "你仍然可以学习更多技能。请在回来前使用所有剩余技能点。";
			close;
		}
		jobchange Job_high_Wizard;
		set ADVJOB,0;
		mes "[进阶巫师]";
		mes "恭喜。";
		mes "作为进阶巫师，我希望你使用力量为受压迫者带来和平。";
		close;
	}
	mes "[进阶巫师]";
	mes "欢迎来到英灵殿——荣誉大厅。";
	next;
	mes "[进阶巫师]";
	mes "请在这里安心休息。";
	mes "向勇士致敬！";
	close;
}
