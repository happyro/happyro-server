valkyrie,44,42,5	script	High Priest#Valkyrie	60,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[进阶牧师]";
			mes "恭喜。";
			mes "向战士致敬！";
			close;
		}
		mes "[进阶牧师]";
		mes "借助神圣之力，愿我们找到和平、力量与保护。请将我们从邪恶势力中拯救出来……";
		close;
	}
	else if (ADVJOB == Job_High_Priest && Class == Job_Acolyte_High && JobLevel > 39) {
		mes "[进阶牧师]";
		mes "我们的世界需要有才华且坚定的人。";
		mes "请继续你的善行，成为更加伟大的圣洁英雄……";
		next;
		mes "[进阶牧师]";
		mes "你愿意成为进阶牧师吗？";
		next;
		if (select("不:是") == 1) {
			mes "[进阶牧师]";
			mes "准备好后随时回来。";
			mes "向战士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[进阶牧师]";
			mes "你还可以学习更多技能。请在回来前用完所有剩余技能点。";
			close;
		}
		jobchange Job_High_Priest;
		set ADVJOB,0;
		mes "[进阶牧师]";
		mes "恭喜。";
		mes "作为进阶牧师，希望你能引导他人走上圣洁之路……";
		close;
	}
	mes "[进阶牧师]";
	mes "欢迎来到英灵殿——荣誉大厅。";
	next;
	mes "[进阶牧师]";
	mes "请在这里安心休息。";
	mes "向战士致敬！";
	close;
}
