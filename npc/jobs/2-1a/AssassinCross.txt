	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[进阶刺客]";
			mes "恭喜……";
			next;
			mes "[进阶刺客]";
			mes "...";
			next;
			mes "[进阶刺客]";
			mes "...";
			mes "......";
			next;
			mes "[进阶刺客]";
			mes "...";
			mes "......";
			mes "向战士致敬。";
			close;
		}
		mes "[进阶刺客]";
		mes "我们是沙漠的战士。没人敢小看我们。";
		mes "没人……";
		close;
	}
	else if (ADVJOB == Job_Assassin_Cross && Class == Job_Thief_High && JobLevel > 39) {
		mes "[进阶刺客]";
		mes "时刻到了。";
		mes "世界需要你……";
		mes "比以往更需要。";
		next;
		mes "[进阶刺客]";
		mes "我希望你继续生活在阴影中，但以全新的外表成为更强大的刺客。";
		next;
		mes "[进阶刺客]";
		mes "你愿意成为进阶刺客吗？";
		next;
		if (select("不:是") == 1) {
			mes "[进阶刺客]";
			mes "准备好后再回来。";
			next;
			mes "[进阶刺客]";
			mes "向战士致敬。";
			close;
		}
		if (SkillPoint) {
			mes "[进阶刺客]";
			mes "你还没有学会所有能学的技能。";
			next;
			mes "[进阶刺客]";
			mes "用完所有技能点再回来。";
			close;
		}
		jobchange Job_Assassin_Cross;
		set ADVJOB,0;
		mes "[进阶刺客]";
	mes "恭喜你。";
		mes "作为进阶刺客，希望你能在黑暗中为更光明的未来而战。";
		close;
	}
	mes "[进阶刺客]";
	mes "欢迎来到英灵殿——荣誉大厅。";
	next;
	mes "[进阶刺客]";
	mes "请在这里安心休息。";
	mes "向战士致敬。";
	close;
}
