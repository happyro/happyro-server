valkyrie,44,50,5	script	MasterSmith#Valkyrie	731,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[进阶铁匠]";
			mes "恭喜。";
			mes "向战士致敬！";
			close;
		}
		mes "[进阶铁匠]";
		mes "啊……";
		mes "工艺的巅峰。";
		mes "那就是进阶铁匠的作品。";
		next;
		mes "[进阶铁匠]";
		mes "当你成为最优秀中的最优秀者，你会开始锻造，目标是发现更新更好的制作方法……";
		close;
	}
	else if (ADVJOB == Job_WhiteSmith && Class == Job_Merchant_High && JobLevel > 39) {
		mes "[进阶铁匠]";
		mes "时刻到了！";
		mes "我们的世界需要勇敢、勤奋的冒险者，像你一样……";
		next;
		mes "[进阶铁匠]";
		mes "你愿意成为进阶铁匠吗？";
		next;
		if (select("不:是") == 1) {
			mes "[进阶铁匠]";
			mes "准备好后随时回来。";
			mes "向战士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[进阶铁匠]";
			mes "你还可以学习更多技能。请在回来前用完所有剩余技能点。";
			close;
		}
		jobchange Job_Whitesmith;
		set ADVJOB,0;
		mes "[进阶铁匠]";
		mes "恭喜！";
		mes "作为进阶铁匠，希望你能为米德加尔特锻造通往更光明未来的道路。";
		close;
	}
	mes "[进阶铁匠]";
	mes "欢迎来到英灵殿——荣誉大厅。";
	next;
	mes "[进阶铁匠]";
	mes "请在这里安心休息。";
	mes "向战士致敬！";
	close;
}
