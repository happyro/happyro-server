valkyrie,44,39,5	script	进阶骑士#Valkyrie::Lord Knight#Valkyrie	56,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[进阶骑士]";
			mes "恭喜你。";
			mes "向勇士致敬！";
			close;
		}
		mes "[进阶骑士]";
		mes "我们骑士肩负着重大的责任……";
		mes "那就是服务与守护。";
		next;
		mes "[进阶骑士]";
		mes "即使要付出自己的生命，我们也必须守护同伴的安危。";
		close;
	}
	else if (ADVJOB == Job_Lord_Knight && Class == Job_Swordman_High && JobLevel > 39) {
		mes "[进阶骑士]";
		mes "你的时刻到了！";
		mes "这个世界仍然需要你。";
		mes "请以全新的姿态继续英雄的人生。";
		next;
		mes "[进阶骑士]";
		mes "你想成为进阶骑士吗？";
		next;
		if (select("不。:是。") == 1) {
			mes "[进阶骑士]";
			mes "准备好后，随时欢迎回来。";
			mes "向勇士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[进阶骑士]";
			mes "你仍然可以学习更多技能。请在回来找我之前，使用完所有剩余的技能点。";
			close;
		}
		jobchange Job_Lord_Knight;
		set ADVJOB,0;
		mes "[进阶骑士]";
		mes "恭喜你！";
		mes "作为进阶骑士，我希望你能站在战斗最前线，带领盟友走向胜利！";
		close;
	}
	mes "[进阶骑士]";
	mes "欢迎来到英灵殿——荣誉大厅。";
	next;
	mes "[进阶骑士]";
	mes "请在这里放松休息。";
	mes "向勇士致敬！";
	close;
}
