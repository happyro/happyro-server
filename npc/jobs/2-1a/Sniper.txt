valkyrie,44,55,5	script	进阶猎人#Valkyrie::Sniper#Valkyrie	727,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[进阶猎人]";
			mes "恭喜。";
			mes "向勇士致敬！";
			close;
		}
		mes "[进阶猎人]";
		mes "一箭。";
		mes "一击必杀。";
		mes "一旦培养出这种战斗方式的视野，就没有那么难了。";
		close;
	}
	else if (ADVJOB == Job_Sniper && Class == Job_Archer_High && JobLevel > 39) {
		mes "[进阶猎人]";
		mes "这个世界需要像您这样的强大弓箭手。";
		mes "您准备好承担这份重大责任了吗？";
		next;
		mes "[进阶猎人]";
		mes "您愿意迈出下一步，成为一名进阶猎人吗？";
		next;
		if (select("不。:是。") == 1) {
		mes "[进阶猎人]";
			mes "准备好后，随时欢迎回来。";
			mes "向勇士致敬！";
			close;
			}
		if (SkillPoint) {
		mes "[进阶猎人]";
			mes "您仍然可以学习更多技能。请在回来找我之前用完所有剩余技能点。";
			close;
			}
		jobchange Job_Sniper;
		set ADVJOB,0;
		mes "[进阶猎人]";
		mes "恭喜！";
		mes "成为进阶猎人后，只要邪恶爪牙出现在您的视线中，它们就永远不得安宁！";
		close;
		}
	mes "[进阶猎人]";
	mes "欢迎来到英灵殿——荣誉大厅。";
	next;
	mes "[进阶猎人]";
	mes "请在这里好好休息。";
	mes "向勇士致敬！";
	close;
	}
