	.@owner = getvariableofnpc(.Owner,"Monster Summon#cr0");
	.@account = getvariableofnpc(.Account,"Monster Summon#cr0");
	if (isloggedin(.@account,.@owner) && getmapxy(.@map$,.@x,.@y,BL_PC,.@account) == 0 && .@map$ == "job_cru" && .@x >= 160 && .@x <= 175 && .@y >= 14 && .@y <= 183 && attachrid(.@account)) {
		if (CRUS_Q == 10)
			warp "prt_cas",238,94;
		else
			warp "job_cru",24,169;
		detachrid;
	}
	donpcevent "Monster Summon#cr0::OnReset";
	end;

	end;

	end;

	end;
