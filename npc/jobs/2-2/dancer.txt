	waitingroom "等候室",20,"Waiting Room#dance::OnStartArena",1;
	donpcevent "Waiting Room#dance::OnReset";
	end;

OnStartArena:
	if (.Active) end;
	disablewaitingroomevent;
	.Active = 1;
	.Round++;
	warpwaitingpc "job_duncer",69,110,1;
	if (!$@warpwaitingpcnum || !attachrid($@warpwaitingpc[0])) {
		donpcevent "Waiting Room#dance::OnReset";
		end;
	}
	if (DANC_Q != 8) {
		warp "comodo",188,162;
		donpcevent "Waiting Room#dance::OnReset";
		end;
	}
	.Owner = getcharid(0);
	.Account = getcharid(3);
	detachrid;
	donpcevent "Bijou#dance_timer::OnEnable";
	end;

OnFinish:
	if (!.Active) end;
	if (isloggedin(.Account,.Owner)) {
		getmapxy .@map$,.@x,.@y,BL_PC,.Account;
		if (.@map$ == "job_duncer" && ((.@x >= 68 && .@x <= 70 && .@y >= 106 && .@y <= 114) || (.@x >= 65 && .@x <= 73 && .@y >= 109 && .@y <= 111))) {
			if (attachrid(.Account)) {
				if (DANC_Q == 8) {
					DANC_Q = 9;
					changequest 7005,7006;
					mapannounce "job_duncer","很好！做得漂亮！回去找比茹！",bc_map;
				}
				warp "comodo",188,162;
				detachrid;
			}
		}
	}
	donpcevent "Waiting Room#dance::OnReset";
	end;

OnReset:
	.Active = 0;
	.Owner = 0;
	.Account = 0;
	donpcevent "Bijou#dance_timer::OnDisable";
	disablenpc "dance#up";
	disablenpc "dance#down";
	disablenpc "dance#left";
	disablenpc "dance#right";
	disablenpc "dance#cen";
	donpcevent "dance#poring::OnDisable";
	enablewaitingroomevent;
	end;
}
