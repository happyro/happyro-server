-	script	菲尼克斯::PhoenixTKJobChange	-1,{
	if(Upper == 2) {
		mes "[菲尼克斯]";
		mes "你好，孩子。";
		close;
	} else if(Class == Job_Taekwon) {
		mes "[菲尼克斯]";
		mes "你的训练进行得如何？随着你的技巧变得更加精湛或华丽，永远不要忘记你始终可以依靠基础。";
		close;
	} else if(Class > Job_Novice) || (Class == Job_Novice && TK_Q == 0) {
		mes "[菲尼克斯]";
		mes "这片土地。我们曾经美丽的世界如今已被邪恶污染：有太多人类被黑暗腐化，也有太多怪物威胁无辜者……";
		next;
		mes "[菲尼克斯]";
		mes "这个世界肆虐的灾祸对普通人类来说太过沉重，他们无法独自抵抗如此压倒性的危机。然而，人必须立志挑战逆境。";
		next;
		mes "[菲尼克斯]";
		mes "因此，我致力于让自己变得更强。我一直训练以领悟武道，钻研一种磨炼身心的技艺，并希望与世界分享。";
		next;
		mes "[菲尼克斯]";
		mes "我也许无法独自改变世界，但我不会停止训练自己的精神与身体。我知道答案终会到来……";
		next;
		if (Class != Job_Novice) {
			mes "[菲尼克斯]";
			mes "尊贵的冒险者：如果你认识尚未选择人生道路的人，请把他推荐给我。如果他感兴趣，我可以传授我的技艺……";
			close;
		}
		mes "[菲尼克斯]";
		mes "如果你还没有决定今后要走的人生道路，请考虑成为我的技艺传承者。这并不容易，但会带给你强大的力量……";
		next;
		if(select("好的，我加入你。","不用了，谢谢。") == 1) {
			if(!callfunc("F_CanChangeJob")) {
				mes "[菲尼克斯]";
				mes "真遗憾！以你目前的职业等级，还不足以在我的指导下开始训练。学会 9 级基本技能后再回来。";
				close;
			}
			mes "[菲尼克斯]";
			mes "很好……我接受你成为我的学生。开始训练时，必须先锻炼你的身体，才能施展将要学习的技能。";
			next;
			if(BaseLevel > 19) {
				mes "[菲尼克斯]";
				mes "嗯。我看得出你已经完成了充分的身体训练。很好。现在准备精神训练。深呼吸，准备好后告诉我。";
				set TK_Q, 2;
				setquest 6001;
				close;
			}
			mes "[菲尼克斯]";
			mes "心灵不一定受身体限制，但若不融合身心，就永远无法发挥真正潜力。去提升 ^FF00001 个基础等级^000000，然后回来。";
			next;
			set TAEK_Q, BaseLevel;
			set TK_Q, 1;
			setquest 6000;
			mes "[菲尼克斯]";
			mes "我知道这对新手来说并不容易，但你必须为这份职业的艰辛做好准备。下次见面时，我期待你变得更强。";
			close;
		}
		mes "[菲尼克斯]";
		mes "我明白。人生有许多道路，但你一次只能选择其中一条。我希望你以自己的方式探索武道，冒险者。";
		close;
	}
	switch(TK_Q) {
	case 1:
		if(BaseLevel > TAEK_Q) {
			set TK_Q, 2;
			changequest 6000,6001;
			mes "[菲尼克斯]";
			mes "很好。我感觉到你更加契合自己的内在力量。这是提升等级的自然结果。我们可以继续下一阶段的训练了。";
			close;
		}
		mes "[菲尼克斯]";
		mes "你必须再提升 ^FF00001 个基础等级^000000，以证明你能够承受这份职业带来的艰辛。永远不要忽视训练。";
		close;
	case 2:
		mes "[菲尼克斯]";
		mes "作为精神训练，我会问你一系列问题来考验你的精神。放松。尽可能诚实地回答。你的意志和信念将受到考验。";
		next;
		mes "[菲尼克斯]";
		mes "作为我的技艺传承者，在战斗中迅速做出最佳决定的能力是必需的。现在开始提问。";
		next;
		mes "[菲尼克斯]";
		mes "当你遇到巨大困难时，通常会如何应对？";
		next;
		switch(select( "正面迎击。:想办法避开。:重新整队并分析问题。")) {
		case 1:
			mes "[菲尼克斯]";
			mes "是的，这正是我想要的答案。即使你一开始无法解决问题，我们也能从这种坚定的决心中受益。不要让任何障碍阻挡你。";
			next;
			mes "[菲尼克斯]";
			mes "即使失败，只有当你全力以赴时，才能从经历中学习。敷衍的尝试很少带来有益的结果。";
			next;
			mes "[菲尼克斯]";
			mes "既然你已经理解意志的重要性，我们将继续下一个问题。";
			next;
			break;
		case 2:
			mes "[菲尼克斯]";
			mes "蠢货！逃避挑战，你怎么能成熟？恐惧有时是能救命的健康反应，但真正的懦弱令人鄙夷。";
			next;
			mes "[菲尼克斯]";
			mes "听到你这样说让我失望。不要再对我说这种话。嗯。思考勇气与懦弱的含义，然后再来和我说话。";
			close;
		case 3:
			mes "[菲尼克斯]";
			mes "重新整队？被击败后这样做是有好处的。但最好在遇到问题时就正面面对。你不会总有重新整队的余裕。";
			next;
			mes "[菲尼克斯]";
			mes "问题可以预测并分析，但我认为立即撤退并不明智。思考你的恐惧，以及你定义的失败。然后回来找我。";
			close;
		}
		mes "[菲尼克斯]";
		mes "旅途中，你会遇到许多有着不同背景和观点的人。不可避免地，你会遇到一个无法理解其生活方式的人。";
		next;
		mes "[菲尼克斯]";
		mes "同样，那个人也无法理解你的生活方式。当你们的观点发生冲突并引发激烈争执时，你会如何回应？";
		next;
		switch( select("无论如何坚持我是对的:无视冲突的观点:接受差异并从中学习")) {
		case 1:
			mes "[菲尼克斯]";
			mes "拥有自己的观点很重要。但是，你必须认识到自己可能错了，相反的观点也可能有道理。";
			next;
			mes "[菲尼克斯]";
			mes "不存在唯一正确的答案，真理之光可以呈现多种色彩。自然就是如此。把想法强加给别人是一种压迫行为。";
			next;
			mes "[菲尼克斯]";
			mes "不要把自己限制在单一观点中，也不要因固守单一真理而扼杀成长。思考这一点，然后再来和我谈谈。";
			close;
		case 2:
			mes "[菲尼克斯]";
			mes "与他人相处很重要，但如果没有自己的独特贡献、思想和观点，你无法为世界带来价值。";
			next;
			mes "[菲尼克斯]";
			mes "在尊重自己和他人的前提下，理念上的冲突是拓展视野、理解他人眼中世界的绝佳机会。";
			next;
			mes "[菲尼克斯]";
			mes "思考如何与自我和他人建立和谐。然后回来找我。";
			close;
		case 3:
			mes "[菲尼克斯]";
			mes "很好。你必须看清差异的真实面貌，也要以优雅和真诚的态度接受对自己观点的批评，认真思考。";
			next;
			mes "[菲尼克斯]";
			mes "不可能了解世上的一切。不可能理解每种观点。但这不代表你不理解的观点就毫无价值。";
			next;
			mes "[菲尼克斯]";
			mes "与你意见不同的人可能拥有你不知道的答案。在你脆弱时，这个人可能成为你最大的帮助。记住这一点。";
			next;
		}
		mes "[菲尼克斯]";
		mes "我对你给出的答案很满意。请思考我们讨论的内容。心情平静后，再来和我谈谈。";
		set TK_Q, 3;
		changequest 6001,6002;
		close;
	case 3:
		mes "[菲尼克斯]";
		mes "你感到平静安宁了吗？我会问你一个非常重要的问题。";
		mes "请诚实回答。";
		next;
		mes "[菲尼克斯]";
		mes "^FF0000你是否准备好修习这门武艺，并坚守它所代表的信念？^000000";
		next;
		if( select( "是的。","不。") == 1) {
			mes "[菲尼克斯]";
			mes "很好。你不再只是学生。现在，你被赋予 ^FF0000跆拳道^000000 门徒的力量与责任。";
			next;
			mes "[菲尼克斯]";
			mes "“跆拳”意为“拳打脚踢”，“道”意为“技艺”。这门武术专注于使用拳脚的技巧。";
			next;
			mes "[菲尼克斯]";
			mes "请用这门修行磨炼身心，学习最适合你的技巧。永远不要懈怠训练，或给跆拳道蒙羞。";
			next;
			if (!checkweight(2101,1)) {
				mes "请先为转职奖励预留足够的负重和物品栏空间。";
				close;
			}
			if (Class != Job_Novice || Upper == 2 || !callfunc("F_CanChangeJob")) {
				mes "[菲尼克斯]";
				mes "你当前的职业或基本技能不符合转职条件，请确认后再来。";
				close;
			}
			if (SkillPoint != 0) {
				mes "[菲尼克斯]";
				mes "请先分配完所有技能点，再来完成转职。";
				close;
			}
			callfunc "Job_Change",Job_Taekwon;
			if (isbegin_quest(13101) == 1)
				completequest 13101;
			completequest 6002;
			callfunc "F_ClearJobVar";
			getitem 2101,1; // Guard[0]
			mes "[菲尼克斯]";
			mes "你还年轻，我想你会需要一个职业称号。";
			if(Sex == SEX_FEMALE) {
				mes "既然如此，你现在是 ^FF0000跆拳少女^000000。";
			} else {
				mes "既然如此，你现在是 ^FF0000跆拳少年^000000。";
			}
			mes "是的，听起来不错。";
			next;
			mes "[菲尼克斯]";
			mes "请收下这面盾牌，好好使用这份礼物。旅行和训练时，向他人介绍我们的技艺，也从他们那里学习。";
			next;
			mes "[菲尼克斯]";
			mes "现在是时候踏上自己的旅程，寻找新的挑战来增强力量了。作为跆拳道修行者，请自豪地前行……";
			next;
			mes "[菲尼克斯]";
			mes "很好。祝你好运。希望将来还能再见到你，"+strcharinfo(0)+"。";
			close;
		}
		mes "[菲尼克斯]";
		mes "嗯，也许你还没准备好从学生成为正式门徒。准备好后，再来和我谈谈。";
		close;
	}
}
