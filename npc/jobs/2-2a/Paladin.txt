valkyrie,53,39,3	script	进阶十字军#Valkyrie::Paladin#Valkyrie	752,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[进阶十字军]";
			mes "恭喜。";
			mes "向勇士致敬！";
			close;
		}
		mes "[进阶十字军]";
		mes "别忘了圣战即将到来！";
		mes "我们必须做好准备！";
		mes "愿正义之光永远照亮道路！";
		close;
	}
	else if (ADVJOB == Job_Paladin && Class == Job_Swordman_High && JobLevel > 39) {
		mes "[进阶十字军]";
		mes "圣战将在我们意识到之前降临。比以往更需要强壮的男女为善与正义而战。";
		next;
		mes "[进阶十字军]";
		mes "你愿意作为进阶十字军站在正义一方战斗吗？";
		next;
		if (select("不。:是。") == 1) {
			mes "[进阶十字军]";
			mes "准备好后，随时回来。";
			mes "向勇士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[进阶十字军]";
			mes "你仍然可以学习更多技能。请在回来前使用所有剩余技能点。";
			close;
		}
		jobchange Job_Paladin;
		set ADVJOB,0;
		mes "[进阶十字军]";
		mes "恭喜。";
		mes "作为进阶十字军，我希望你保护比你弱小的人，并为我们带来即将到来的善恶之战的胜利。";
		close;
	}
	mes "[进阶十字军]";
	mes "欢迎来到英灵殿——荣誉大厅。";
	next;
	mes "[进阶十字军]";
	mes "请在这里安心休息。";
	mes "向勇士致敬！";
	close;
}
