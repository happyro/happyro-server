valkyrie,53,58,3	script	进阶流氓#Valkyrie::Stalker#Valkyrie	747,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[进阶流氓]";
			mes "恭喜。";
			mes "向战士致敬！";
			close;
		}
		mes "[进阶流氓]";
		mes "嘿……";
		mes "成为英雄很辛苦，还要阴险、不可信、偷偷摸摸……";
		next;
		mes "[进阶流氓]";
		mes "但当事情变得艰难时，我的伙伴知道可以依靠我。";
		mes "我需要他们，他们也需要我。";
		close;
	}
	else if (ADVJOB == Job_Stalker && Class == Job_Thief_High && JobLevel > 39) {
		mes "[进阶流氓]";
		mes "这个世界需要更多愿意在秩序与无法无天之间走钢丝的英雄。";
		next;
		mes "[进阶流氓]";
		mes "你准备好加入最狡猾的战士行列吗？你准备好成为进阶流氓了吗？";
		next;
		if (select("不:是") == 1) {
			mes "[进阶流氓]";
			mes "准备好后随时回来。";
			mes "向战士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[进阶流氓]";
			mes "你还可以学习更多技能。请在回来前用完所有剩余技能点。";
			close;
		}
		jobchange Job_Stalker;
		set ADVJOB,0;
		mes "[进阶流氓]";
		mes "恭喜！";
		mes "作为进阶流氓，希望你刺中正确的人的后背，用他们自己的卑鄙手段消灭邪恶！";
		close;
	}
	mes "[进阶流氓]";
	mes "欢迎来到英灵殿——荣誉大厅。";
	next;
	mes "[进阶流氓]";
	mes "请在这里安心休息。";
	mes "向战士致敬！";
	close;
}
