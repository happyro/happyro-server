valkyrie,53,50,3	script	进阶炼金术师#Valkyrie::Biochemist#Valkyrie	122,{
	if (Class == Job_Creator && bioeth == 13 && getskilllv("AM_BIOETHICS") == 0) {
		mes "[进阶炼金术师]";
		mes "啊，你是来找回失去的记忆吗？是的，你一定是为了曾经属于你的生命秘密而来……";
		next;
		if (select("是:否") == 1) {
			mes "[进阶炼金术师]";
			mes "闭上眼睛，放松心神。";
			mes "我们将回到你的过去，重新收集失落记忆的碎片。";
			next;
			mes "[进阶炼金术师]";
			mes "睁开眼睛后，你会清楚记得生命的秘密，也会记得使用这些秘密时肩负的责任，并将它们用于正途……";
			next;
			skill "AM_BIOETHICS",1,SKILL_PERM;
			mes "[进阶炼金术师]";
			mes "睁开眼睛……";
			mes "现在你已经记起如何创造人工生命，我只希望你尊重你所有的创造物。";
			close;
		}
		mes "[进阶炼金术师]";
		mes "如果你想找回失去的记忆，请回来找我。";
		mes "创造生命的秘密绝非小事……";
		close;
	}
	else if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[进阶炼金术师]";
			mes "恭喜。";
			mes "向勇士致敬！";
			close;
		}
		mes "[进阶炼金术师]";
		mes "像我这样的人出现在这里很奇怪。";
		mes "但即使精通科学之道的人也能成为英雄。";
		next;
		mes "[进阶炼金术师]";
		mes "在这种情况下，重要的不一定是我使用的手段，而是我为之奋斗的目标，它们让我赢得英灵殿的一席之地……";
		close;
	}
	else if (ADVJOB == Job_Creator && Class == Job_Merchant_High && JobLevel > 39) {
		mes "[进阶炼金术师]";
		mes "是的……";
		mes "是时候了。";
		mes "米德加尔特需要更多像你这样的天才。";
		next;
		mes "[进阶炼金术师]";
		mes "你想成为进阶炼金术师吗？";
		next;
		if (select("不。:是。") == 1) {
			mes "[进阶炼金术师]";
			mes "准备好后，随时回来。";
			mes "向勇士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[进阶炼金术师]";
			mes "你仍然可以学习更多技能。请在回来前使用所有剩余技能点。";
			close;
		}
		jobchange Job_Creator;
		if (bioeth == 13)
			skill "AM_BIOETHICS",1,SKILL_PERM;
		set ADVJOB,0;
		mes "[进阶炼金术师]";
		mes "恭喜！";
		mes "作为进阶炼金术师，希望你将渊博的知识用于正确的目的。";
		mes "不要辜负这份力量。";
		close;
	}
	else {
		mes "[进阶炼金术师]";
		mes "欢迎来到英灵殿——荣誉大厅。";
		next;
		mes "[进阶炼金术师]";
		mes "请在这里安心休息。";
		mes "英灵殿欢迎每一位来客。";
		mes "向勇士致敬！";
		close;
	}
}
