valkyrie,53,42,3	script	Champion#Valkyrie	52,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[进阶武僧]";
			mes "恭喜。";
			mes "向勇士致敬！";
			close;
		}
		mes "[进阶武僧]";
		mes "技能。";
		mes "速度。";
		mes "力量。";
		mes "敏捷。";
		next;
		mes "[进阶武僧]";
		mes "进阶武僧可以从这些方面获益。";
		mes "但是，一个人一生只能掌握这么多……";
		close;
	}
	else if (ADVJOB == Job_Champion && Class == Job_Acolyte_High && JobLevel > 39) {
		mes "[进阶武僧]";
		mes "时机到了。";
		mes "伟大的英雄们该站出来，对抗肆虐米德加尔特世界的邪恶势力了！";
		next;
		mes "[进阶武僧]";
		mes "您愿意成为进阶武僧吗？";
		next;
		if (select("不。:是。") == 1) {
			mes "[进阶武僧]";
			mes "准备好后，随时欢迎回来。";
			mes "向勇士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[进阶武僧]";
			mes "您仍然可以学习更多技能。请在回来找我之前用完所有剩余技能点。";
			close;
		}
		jobchange Job_Champion;
		set ADVJOB,0;
		mes "[进阶武僧]";
		mes "恭喜！";
		mes "作为进阶武僧生活，用双拳的力量为这个世界带来光明。";
		close;
	}
	mes "[进阶武僧]";
	mes "欢迎来到英灵殿——荣誉大厅。";
	next;
	mes "[进阶武僧]";
	mes "请在这里好好休息。";
	mes "向勇士致敬！";
	close;
	}
