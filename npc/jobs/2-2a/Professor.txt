valkyrie,53,47,3	script	Scholar#Valkyrie	743,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[进阶贤者]";
			mes "恭喜你。";
			mes "向勇士致敬！";
			close;
		}
		mes "[进阶贤者]";
		mes "这需要一生……";
		mes "确实需要一生的时间才能积累成为进阶贤者所需的知识……";
		next;
		mes "[进阶贤者]";
		mes "这令人不知所措。";
		mes "你学得越多，就越会发现自己不知道的事情。";
		mes "学习的过程永无止境……";
		close;
	}
	else if (ADVJOB == Job_Professor && Class == Job_Mage_High && JobLevel > 39) {
		mes "[进阶贤者]";
		mes "米德加尔特的进阶贤者还不够多，无法帮助开启繁荣的新时代。";
		mes "这个世界需要你……";
		next;
		mes "[进阶贤者]";
		mes "你愿意承担这份重大的责任吗？";
		mes "你愿意作为进阶贤者为米德加尔特效力吗？";
		next;
		if (select("不。:是。") == 1) {
			mes "[进阶贤者]";
			mes "准备好后，随时欢迎回来。";
			mes "向勇士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[进阶贤者]";
			mes "你仍然可以学习更多技能。请在回来找我之前，使用完所有剩余的技能点。";
			close;
		}
		jobchange Job_Professor;
		set ADVJOB,0;
		mes "[进阶贤者]";
		mes "恭喜你！";
		mes "作为进阶贤者，我希望你能积极参与将知识的光明带到黑暗之处。";
		close;
	}
	mes "[进阶贤者]";
	mes "欢迎来到英灵殿——荣誉大厅。";
	next;
	mes "[进阶贤者]";
	mes "请在这里放松休息。";
	mes "向勇士致敬！";
	close;
}
