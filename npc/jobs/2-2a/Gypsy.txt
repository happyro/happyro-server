valkyrie,53,56,3	script	Gypsy#Valkyrie	101,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[进阶舞娘]";
			mes "恭喜你。";
			mes "向勇士致敬！";
			close;
		}
		mes "[进阶舞娘]";
		mes "向左走，";
		mes "向右走～！";
		mes "再踏一步……！";
		mes "跳舞不只是一种爱好。";
		mes "对我来说，这是生活方式～";
		close;
	}
	else if (ADVJOB == Job_Gypsy && Class == Job_Archer_High && JobLevel > 39) {
		mes "[进阶舞娘]";
		mes "米德加尔特大陆需要有才华的女性，巧妙地改变善与恶之战的平衡。";
		next;
		mes "[进阶舞娘]";
		mes "你准备好承担这个职责，成为进阶舞娘了吗？";
		next;
		if (select("不。:是。") == 1) {
			mes "[进阶舞娘]";
			mes "准备好后，随时欢迎回来。";
			mes "向勇士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[进阶舞娘]";
			mes "你仍然可以学习更多技能。请在回来找我之前，使用完所有剩余的技能点。";
			close;
		}
		jobchange Job_Gypsy;
		set ADVJOB,0;
		mes "[进阶舞娘]";
		mes "恭喜你！";
		mes "作为进阶舞娘，我知道你的表演会打动所有观看者的心……";
		close;
	}
	mes "[进阶舞娘]";
	mes "欢迎来到英灵殿——荣誉大厅。";
	next;
	mes "[进阶舞娘]";
	mes "请在这里放松休息。";
	mes "向勇士致敬！";
	close;
}
