valkyrie,53,54,3	script	Minstrel#Valkyrie	741,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[进阶诗人]";
			mes "恭喜。";
			mes "向勇士致敬！";
			close;
		}
		mes "[进阶诗人]";
		mes "你想和我一起唱歌吗？";
		mes "沙啦啦啦啦~";
		close;
	}
	else if (ADVJOB == Job_Clown && Class == Job_Archer_High && JobLevel > 39) {
		mes "[进阶诗人]";
		mes "沉闷的凡人世界需要更多欢快的歌曲。";
		mes "你愿意为他们带去歌声，扭转对抗邪恶的战局吗？";
		next;
		mes "[进阶诗人]";
		mes "你愿意为米德加尔特这样做吗……";
		mes "以进阶诗人的身份？";
		next;
		if (select("不。:是。") == 1) {
			mes "[进阶诗人]";
			mes "准备好后，随时回来。";
			mes "向勇士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[进阶诗人]";
			mes "你仍然可以学习更多技能。请在回来前使用所有剩余技能点。";
			close;
		}
		jobchange Job_Clown;
		set ADVJOB,0;
		mes "[进阶诗人]";
		mes "恭喜！";
		mes "作为进阶诗人，你的歌曲会给盟友带来希望，给敌人带来绝望。";
		close;
	}
	mes "[进阶诗人]";
	mes "欢迎来到英灵殿——荣誉大厅。";
	next;
	mes "[进阶诗人]";
	mes "请在这里安心休息。";
	mes "向勇士致敬！";
	close;
}
