y_airport,143,43,5	script	机场工作人员#y_air1a::y_airport1	90,{
	mes "[机场工作人员]";
	mes "欢迎来到朱诺机场。这里提供前往艾因布洛克、里希塔乐和休尔格尔的国内航班，以及前往伊兹鲁德和拉赫尔的国际航班。";
	mes "有什么可以为您服务的吗？";
	next;
	if (select("登上飞艇。:取消。") == 1) {
		mes "[机场工作人员]";
		mes "所有航班的登机费都是 1,200 Zeny。";
		mes "如果您使用飞空艇免费券，登机费将被免除。";
		mes "要出发吗？";
		next;
		if (select("是:否") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "y_airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "y_airport",148,51;
				end;
			}
			mes "[机场工作人员]";
			mes "抱歉，您没有足够的 Zeny 支付 1,200 Zeny 的登机费。";
			mes "请准备好费用。";
			close;
		}
	}
	mes "[机场工作人员]";
	mes "谢谢您，祝您今天愉快。";
	close;
}
