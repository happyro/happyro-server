airport,143,43,5	script	机场工作人员#airport1a::airport1	90,{
	mes "[机场工作人员]";
	mes "欢迎来到艾因布洛克机场，这里提供直达朱诺、里希塔乐和休尔格尔的航班。";
	next;
	if (select("登上飞艇:取消") == 1) {
	mes "[机场工作人员]";
		mes "飞艇登乘费为 1,200 Zeny，但如果您有飞空艇免费搭乘券，即可免除费用。要登上飞艇吗？";
		next;
		if (select("是:否") == 1) {
		if (countitem(7311) > 0) {
		delitem 7311,1; //Free_Flying_Ship_Ticket
			warp "airport",148,51;
				end;
				}
				if (Zeny >= 1200) {
			set Zeny, Zeny - 1200;
			warp "airport",148,51;
				end;
				}
				mes "[机场工作人员]";
			mes "抱歉，您没有飞空艇免费搭乘券，且没有足够的 Zeny 支付飞艇登乘费。请记住，登乘费为 1,200 Zeny。";
			close;
			}
			}
			mes "[机场工作人员]";
			mes "谢谢，祝您今天愉快。";
	close;
	}
