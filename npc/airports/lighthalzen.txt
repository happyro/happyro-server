lhz_airport,143,43,5	script	机场工作人员#lhz_air1a::lhz_airport1	90,{
	mes "[机场工作人员]";
	mes "欢迎来到里希塔乐机场，这里提供直达艾因布洛克、朱诺和休尔格尔的航班。";
	next;
	if (select("登上飞艇:取消") == 1) {
		mes "[机场工作人员]";
		mes "登乘费为 1,200 Zeny，但使用飞空艇免费搭乘券即可免除费用。";
		next;
		if (select("是:否") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "lhz_airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "lhz_airport",148,51;
				end;
			}
			mes "[机场工作人员]";
			mes "抱歉，您没有足够的 Zeny 支付 1,200 Zeny 的登乘费用。";
			close;
		}
	}
	mes "[机场工作人员]";
	mes "谢谢，欢迎下次再来。";
	mes "祝您今天愉快～";
	close;
	}
