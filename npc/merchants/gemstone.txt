payon,173,238,5	script	杰依德#pay::Jade#pay	754,{
	if (checkweight(1201,1) == 0) {
		mes "^3355FF等一下！你现在携带的物品太多了。请先把一些物品存入卡普拉仓库后再来。^000000";
		close;
	}
	mes "[" + strnpcinfo(1) + "]";
	mes "带给我两颗相同颜色的宝石，我就能把它们变成另一种颜色的宝石。";
	next;
	switch(select("把蓝色宝石换成红色！:把红色宝石换成黄色！:把黄色宝石换成蓝色！")) {
	case 1: callsub S_TradeGems,717,716;
	case 2: callsub S_TradeGems,716,715;
	case 3: callsub S_TradeGems,715,717;
	}

S_TradeGems:
	.@item_req = getarg(0);
	.@item_id = getarg(1);

	if (countitem(.@item_req) < 2) {
		mes "[" + strnpcinfo(1) + "]";
		mes "哈……！";
		mes "你是在开玩笑吧？";
		mes "如果你不给我至少 2 颗"+ getitemname(.@item_req) +"，我就无法为你提供服务！";
		close;
	}
	else {
		.@gems = countitem(.@item_req) /2;
		mes "[" + strnpcinfo(1) + "]";
		mes "我可以使用你目前拥有的" + getitemname(.@item_req) + "，总共制作 " + .@gems + " 颗" + getitemname(.@item_id) + "。";
		mes "你想怎么做？";
		next;
		switch(select("尽可能多地给我。:我想设定数量。:我不换了。")) {
		case 1:
			delitem .@item_req,.@gems * 2;
			getitem .@item_id,.@gems;
			break;
		case 2:
			mes "[" + strnpcinfo(1) + "]";
			mes "那么你想要多少？请输入你需要的数量，最多为 100。";
			next;
			while(1) {
				input .@input,0,101;
				if (.@input == 0) {
					mes "[" + strnpcinfo(1) + "]";
					mes "一颗都不要？";
					mes "我猜你改变主意了……";
					mes "那就下次再来吧。";
					close;
				}
				else if (.@input > 100) {
					mes "[" + strnpcinfo(1) + "]";
					mes "呃……";
					mes "我让你输入不超过 100 的数量，记得吗……？";
					mes "请重新输入。";
					next;
				}
				else if (.@gems < .@input) {
					// Custom dialogue
					mes "[" + strnpcinfo(1) + "]";
					mes "呃……";
					mes "你没有那么多宝石可以用来兑换……";
					next;
				}
				else break;
			}
			delitem .@item_req,.@input * 2;
			getitem .@item_id,.@input;
			break;
		case 3:
			mes "[" + strnpcinfo(1) + "]";
			mes "当然，没问题。";
			mes "随时欢迎回来。";
			close;
		}
		mes "[" + strnpcinfo(1) + "]";
		mes "给你。";
		mes "随时欢迎回来。";
		mes "嗯，你为什么这样看我？";
		mes "我的脸上有什么吗？";
		close;
	}
}
