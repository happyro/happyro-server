prontera,73,140,0	script	Milk Vendor	86,{
	if (checkweight(1201,1) == 0) {
		mes "^3355FF等一下！^000000";
		mes "我无法为你提供服务，因为你携带了太多物品。";
		mes "把多余物品存入卡普拉仓库后再来~";
		close;
	}
	mes "[牛奶商人]";
	mes "嘿，嘿……";
	mes "如果你带给我 1 个空瓶和 15 Zeny，我会将它们兑换成 1 瓶牛奶。听起来怎么样？";
	next;
	if (select("兑换所有空瓶。:取消") == 1) {
		if (countitem(713) <= 0) {
			mes "[牛奶商人]";
			mes "嘿……";
			mes "你没有任何空瓶。";
			mes "我无法用其他方式给你牛奶……";
			close;
		}
		set .@bottles,countitem(713);
		set .@total_weight,.@bottles * 50;
		set .@total_cost,.@bottles * 15;
		if (Zeny < .@Total_cost) {
			mes "[牛奶商人]";
			mes "哦，哇~！";
			mes "你的 Zeny 不够，无法将所有这些空瓶兑换成牛奶。你至少需要 " + .@total_cost + " Zeny。";
			close;
		}
		if (MaxWeight - Weight < .@total_weight || !checkweight(519,.@bottles)) {
			mes "[牛奶商人]";
			mes "嗯……";
			mes "你能在我给你这些牛奶前，在背包里腾出一些空间吗？";
			close;
		}
		set Zeny, Zeny-.@total_cost;
		delitem 713,.@bottles; //Empty Bottles
		getitem 519,.@bottles; //Milk
		close;
	}
	close;
}
