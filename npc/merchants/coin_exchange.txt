-	script	::merchant_13_2	-1,{
	set .@n$, "[" + strnpcinfo(1) + "]";
	mes .@n$;
	if (isequipped(2782) && ep13_2_rhea == 100) {
		if (!checkweight(1201,1)) {
			mes "看来你携带的东西太多了。";
			mes "不如把一些物品存进仓库再回来？";
			close;
		}
		mes "你好。";
		mes "你想找什么？";
		next;
		explode .@npc_identity_parts$, strnpcinfo(3), "#"; if (.@npc_identity_parts$[0] == "Merchant of Manuk") {
			setarray .@Items[2],12342,12343,12348;
			set .@Book,11019; //Manuk_Selling_Item
			set .@Coin,6080; //Manuk_Coin
			set .@i, select("查看物品说明:马努克的机遇:马努克的勇气:马努克的信仰:取消");
		}
		else {
			setarray .@Items[2],12344,12345,12349;
			set .@Book,11018; //Splendide_Selling_Item
			set .@Coin,6081; //Splendide_Coin
			set .@i, select("查看物品说明:购买捕虫堇果实罐头:购买毒黄蜂的蜜桶:购买飞马的眼泪:什么也不做");
		}
		mes .@n$;
		if (.@i == 1) {
			mes "这是物品说明。";
			close2;
			readbook .@Book,1;
			end;
		}
		else if (.@i < 5) {
			mes "这种"+getitemname(.@Items[.@i])+"，支付^3131FF1枚硬币^000000就能买到 3 个。";
			next;
			switch(select("购买:不购买")) {
			case 1:
				if (countitem(.@Coin)) {
					mes .@n$;
					mes "感谢光临。";
					if (!checkweight(.@Items[.@i],3)) {
						mes "请先为兑换物品预留足够的负重和物品栏空间。";
						close;
					}
					delitem .@Coin,1; 
					getitem .@Items[.@i],3;
					close;
				}
				else {
					mes .@n$;
					mes "你的硬币不够。";
					close;
				}
			case 2:
				mes .@n$;
				break;
			}
		}
		mes "改变主意后欢迎再来。";
		close;
	}
	else {
		explode .@npc_identity_parts$, strnpcinfo(3), "#"; if (.@npc_identity_parts$[0] == "Merchant of Manuk") {
			mes "Rtt od d";
			mes "Qwo hd is d irr";
		}
		else {
			mes "BurWehAla";
			mes "tasnarAndu Ie Ru";
		}
		close;
	}
}
