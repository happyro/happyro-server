alde_alche,24,188,3	script	Guild Dealer	740,{
	if (checkweight(1201,1) == 0) {
		mes "- 等一下！-";
		mes "- 你当前携带了太多物品。-";
		mes "- 请先把物品存入卡普拉仓库后再来。-";
		close;
	}
	mes "[盖弗·阿尔·夏普]";
	mes "欢迎来到炼金术士联盟。";
	mes "今天我能为你做什么？";
	next;
	switch(select("购买材料:购买制作手册:退出")) {
	case 1:
		mes "[盖弗·阿尔·夏普]";
		mes "你想要什么？";
		next;
		if (select("药钵 - 8 Zeny:取消") == 1) {
			mes "[盖弗·阿尔·夏普]";
			mes "你想要多少？";
			mes "想退出请输入‘0’。";
			next;
			while(1) {
				input .@input,0,2001;
				if (.@input == 0) {
					mes "[盖弗·阿尔·夏普]";
					mes "交易已取消。";
					mes "下次再来。";
					close;
				}
				else if (.@input < 1 || .@input > 2000) {
					mes "[盖弗·阿尔·夏普]";
					mes "数量必须在 1 到 2000 之间。";
					next;
				}
				else {
					break;
				}
			}
			set .@sell,.@input * 8;
			if (Zeny < .@sell) {
				mes "[盖弗·阿尔·夏普]";
				mes "你没有足够的 Zeny。";
				mes "请先确认你有多少 Zeny。";
				close;
			}
			if (!checkweight(7134,.@input)) {
				mes "[盖弗·阿尔·夏普]";
				mes "看起来你无法携带全部物品。";
				mes "请检查物品栏空间。";
				close;
			}
			set Zeny, Zeny-.@sell;
			getitem 7134,.@input; //Medicine_Bowl
			mes "[盖弗·阿尔·夏普]";
			mes "谢谢。";
			mes "下次再来。";
			close;
		}
		mes "[盖弗·阿尔·夏普]";
		mes "那么，需要购买炼金术相关材料时再来吧，好吗？";
		close;
	case 2:
		mes "[盖弗·阿尔·夏普]";
		mes "你需要什么？";
		mes "手册通常售价 100,000 Zeny，但有几本特殊手册价格更高。";
		next;
		switch(select("药水制作指南:酒精制作指南:手榴弹制作指南:酸瓶制作指南:植物瓶制作指南:海星炸弹制作指南:闪耀外套制作指南:浓缩药水制作指南:取消交易")) {
		case 1:
			callsub S_SellManual,7144,100000; //Normal_Potion_Book
			break;
		case 2:
			callsub S_SellManual,7127,100000; //Alcol_Create_Book
			break;
		case 3:
			callsub S_SellManual,7128,100000; //FireBottle_Create_Book
			break;
		case 4:
			callsub S_SellManual,7129,100000; //Acid_Create_Book
			break;
		case 5:
			callsub S_SellManual,7130,100000; //Plant_Create_Book
			break;
		case 6:
			callsub S_SellManual,7131,100000; //Mine_Create_Book
			break;
		case 7:
			callsub S_SellManual,7132,100000; //Coating_Create_Book
			break;
		case 8:
			callsub S_SellManual,7133,240000; //Slim_Potion_Create_Book
			break;
		case 9:
			mes "[盖弗·阿尔·夏普]";
			mes "那么……";
			mes "如果你需要购买制作手册，再回来。";
			close;
		}
	case 3:
		mes "[盖弗·阿尔·夏普]";
		mes "好的，";
		mes "祝你今天愉快。";
		close;
	}

S_SellManual:
	.@item_id = getarg(0);
	.@zeny_req = getarg(1);

	if (Zeny < .@zeny_req) {
		mes "[盖弗·阿尔·夏普]";
		mes "你没有足够的 Zeny。";
		mes "请先确认你有多少 Zeny。";
		close;
	}
	mes "[盖弗·阿尔·夏普]";
	mes getitemname(.@item_id) +"？";
	mes "价格是 "+ .@zeny_req +" Zeny。";
	next;
	if (select("购买:退出") == 1) {
		if (!checkweight(.@item_id,1)) {
			mes "[盖弗·阿尔·夏普]";
			mes "请先为这本手册预留足够的负重和物品栏空间。";
			close;
		}
		Zeny = Zeny - .@zeny_req;
		getitem .@item_id,1;
		mes "[盖弗·阿尔·夏普]";
		mes "感谢你的惠顾。";
		close;
	}
	mes "[盖弗·阿尔·夏普]";
	mes "下次再来。";
	close;
}
