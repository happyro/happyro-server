prontera,123,102,5	script	Merchant#pron	85,{
	if (MaxWeight - Weight < 6301 || checkweight(1201,1) == 0) {
	mes "[商人]";
	mes "哈哈！";
	mes "你是超人吗？";
	mes "你带了这么多东西！";
	mes "最好把一部分存进卡普拉仓库~";
		close;
	}
	mes "[商人]";
	mes "你好！";
	mes "你收集了贝壳或绵绒之类的物品吗？杰勒比结晶呢？对了，我有用。";
	next;
		mes "[商人]";
		mes "当然，我不会要求你免费给我。用 1 瓶红色药水换 5 个贝壳、10 个绵绒或 10 个杰勒比结晶，怎么样？";
	next;
	switch(select("当然！:这也太贵了！:不用了，谢谢。")) {
	case 1:
			mes "[商人]";
			mes "那么你想带给我哪种物品？";
		mes "贝壳、绵绒还是杰勒比结晶？";
		next;
		switch(select("贝壳:杰勒比结晶:绵绒:取消")) {
		case 1: callfunc "F_PotExchange",935,5;
		case 2: callfunc "F_PotExchange",909,10;
		case 3: callfunc "F_PotExchange",914,10;
		case 4:
			mes "[商人]";
			mes "好的～";
			mes "没问题。";
			close;
		}
	case 2:
		mes "[商人]";
		mes "敲诈……？";
		mes "如果你查查交易物品的市场价值，其实是我在这里被敲诈。";
		close;
	case 3:
		mes "[商人]";
		mes "好吧，";
		mes "如果改变主意再回来。";
		mes "随时可以来找我。";
		close;
	}
}

function	script	F_PotExchange	{
	.@item_req = getarg(0);
	.@req_amount = getarg(1);

	if (countitem(.@item_req) < .@req_amount) {
	mes "嘿，你答应的那些"+ getitemname(.@item_req) +"在哪里？";
	mes "把"+ getitemname(.@item_req) +"给我！";
		close;
	}
	else {
		mes "好，让我检查一下，看看你身上有多少"+ getitemname(.@item_req) +"。嗯……";
		next;
		mes "你共有"+ countitem(.@item_req) +" 个"+ getitemname(.@item_req) +"。";
		mes "我可以给你 " + (countitem(.@item_req)/.@req_amount) +" 瓶红色药水。";
		next;
		mes "你觉得怎么样？";
			mes "成交吗？";
		next;
		if (select("成交。:不成交。") == 1) {
		mes "你知道自己想从这些"+ getitemname(.@item_req) +"中换取多少瓶红色药水，对吧？";
			next;
		mes "现在一次可以交换 1 到 100 瓶红色药水。";
		mes "改变主意的话输入‘0’取消。";
			next;
			while(1) {
				input .@input,0,101;
				if (.@input == 0) {
		mes "什么……？";
		mes "你干吗还来找我？";
	mes "真是优柔寡断，对吧？";
					close;
				}
				else if (.@input > 100) {
		mes "一次不能给你超过 100 瓶红色药水。我们再试一次。";
					next;
				}
				else {
					break;
				}
			}
			.@put_out_item = .@input * .@req_amount;
			if (countitem(.@item_req) < .@put_out_item) {
	mes "糟糕，你输入的数量不对。请重新确认可以交换的红色药水数量。";
				close;
			}
			if (!checkweight(501,.@input)) {
				mes "请先为红色药水预留足够的负重和物品栏空间。";
				close;
			}
	mes "给你！";
	mes "检查一下我给你的红色药水数量，应该没问题。谢谢，这笔交易不错~";
			delitem .@item_req,.@put_out_item; //Shell
			getitem 501,.@input; //Red_Potion
			close;
		}
		mes "嗯~";
		mes "好吧。";
		mes "不过，红色药水对你这样的冒险者不是更有用吗？";
		close;
	}
}
