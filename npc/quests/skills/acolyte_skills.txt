prt_church,173,23,4	script	Cleric	79,{
	mes "[服事克利夫特]";
	mes "啊，兄弟们！照顾迷途羔羊的任务是否让你们疲惫不堪？";
	mes "我来帮助你们。";
	next;
	switch (select("^3355FF‘关于服事的隐藏能力’^000000……:^3355FF‘神圣之光’^000000训练:结束对话")) {
	case 1:
		mes "[服事克利夫特]";
		mes "我们教会的成员会随着成长自然学会技能。";
		mes "当服事接近资深阶段时，便能掌握这项特殊技能。";
		mes "这就是我们所说的能力。";
		next;
		mes "[服事克利夫特]";
		mes "面对巨大的危险和与邪恶无休止的斗争，我们的同伴非常需要帮助。";
		mes "^3355FF‘神圣之光’^000000就是那项技能。";
		mes "要获得这项能力，还需要付出一些努力。";
		next;
		mes "[服事克利夫特]";
		mes "要获得技能，你必须找到这些物品。";
		mes "^FF33551 个白宝石^000000";
		mes "^FF33551 个水灵矿石^000000";
		mes "^FF33551 个念珠^000000 ";
		next;
		mes "[服事克利夫特]";
		mes "想要使用这项技能，必须拥有足够的经验。";
		mes "职业等级达到 ^FF335530^000000 才能学习这项技能。";
		next;
		mes "[服事克利夫特]";
		mes "至于牧师和武僧，由于他们已经拥有丰富经验，任何职业等级都能获得这项技能。";
		close;
	case 2:
		if (getskilllv("AL_HOLYLIGHT") == 1) {
			mes "[服事克利夫特]";
			mes "兄弟，你已经拥有‘神圣之光’技能了。";
			mes "你无法再次获得已经拥有的技能……";
			mes "愿你将这项技能用于行善……";
			close;
		}
		if ((countitem(727) > 0) && (countitem(991) > 0) && (countitem(2608) > 0) && (JobLevel > 29 || (BaseJob == Job_Priest || BaseJob == Job_Monk)) && (BaseClass == Job_Acolyte)) {
			mes "[服事克利夫特]";
			mes "你的信仰已经证明你值得获得‘神圣之光’技能。";
			mes "你的能力足以使用这项技能。";
			mes "请明智地使用它……";
			next;
			delitem 727,1; //White_Jewel
			delitem 991,1; //Crystal_Blue
			delitem 2608,1; //Rosary
			skill "AL_HOLYLIGHT",1,SKILL_PERM;
			mes "[服事克利夫特]";
			mes "你现在学会了‘神圣之光’。";
			mes "愿你只将这项技能用于正道……";
			close;
		}
		mes "[服事克利夫特]";
		mes "哦，很明显……";
		mes "你还没有准备好接受‘神圣之光’技能。";
		next;
		mes "[服事克利夫特]";
		mes "你应该仔细听清学习这项技能所需的一切。";
		mes "认真聆听，你就能知道自己的不足，从而不断提升自己。";
		close;
	case 3:
		mes "[服事克利夫特]";
		mes "……";
		mes "我理解你的热忱。";
		mes "你还有很多时间练习并积累经验。";
		mes "愿祝福与你同在……";
		close;
	}
}
