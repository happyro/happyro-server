prontera,113,294,3	script	告示牌#god	837,{
	mes "======== 诸神封印状态 ========";
	setarray .@seal$[1], "斯雷普尼尔","米迦勒之腰带","布里辛嘉曼","雷神之锤";
	for(set .@i,1; .@i<=4; set .@i,.@i+1) {
		set .@val, getd("$God"+.@i);
		if (.@val == 0)
			set .@status$,"未发现";
		else if (.@val < $@god_check1)
			set .@status$,"进行中";
		else if (.@val < $@god_check2)
			set .@status$,"已出现";
		else
			set .@status$,"已解放";
		mes .@seal$[.@i]+" 封印："+.@status$;
	}
	mes " ";
	mes "======= 你的封印状态 ========";
	if (god_sl_1 < 51)
		mes "^ff0000斯雷普尼尔 封印：未完成^000000";
	else
		mes "^00ff00斯雷普尼尔 封印：已完成^000000";
	if (god_eremes < 28)
		mes "^ff0000米迦勒之腰带 封印：未完成^000000";
	else
		mes "^00ff00米迦勒之腰带 封印：已完成^000000";
	if (god_brising < 50)
		mes "^ff0000布里辛嘉曼 封印：未完成^000000";
	else
		mes "^00ff00布里辛嘉曼 封印：已完成^000000";
	if (god_mjo_0 < 11)
		mes "^ff0000雷神之锤 封印：未完成^000000";
	else
		mes "^00ff00雷神之锤 封印：已完成^000000";
	close;
}
