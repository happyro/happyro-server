// Edgar's Offer :: iz_npc
//============================================================
-	script	::Edgar_izlude	-1,{
	if (MISC_QUEST & 16) {
		mes "[埃德加]";
		mes "您又要去艾尔贝塔吗？我给您和上次一样的折扣，只收 250 Zeny。";
		mes "听起来怎么样？";
		next;
		if (select("好的～！:您为什么对我这么好？！") == 1) {
			if (Zeny < 250) {
				mes "[埃德加]";
				mes "嗯……";
				mes "这点钱不够。";
				mes "您还是去多赚点钱吧？";
				close ;
			}
			set Zeny, Zeny-250;
			warp "alberta",195,164;
			end;
		}
		mes "[埃德加]";
		mes "我就是这样的人。而且您淘气的笑容让我想起心爱的金发儿子，他多年前离家驾驶自己的船成为水手。愿上天保佑他，无论他身在何方。";
		close;
	}
	mes "[埃德加]";
	mes "我的城镇依斯鲁得通过西边的港口与艾尔贝塔相连。两地往来如此频繁，我都快成艾尔贝塔人了。呵呵呵～";
	next;
	mes "[埃德加]";
	mes "我很熟悉的 菲利克斯 住在艾尔贝塔。那家伙非常吝啬……什么都要收费！";
	next;
	mes "[埃德加]";
	mes "不过他人很好，喜欢帮助别人。只要您付得起价钱，他就会热心提供帮助。";
	next;
	mes "[埃德加]";
	mes "最近艾尔贝塔的人说他其实是在帮助大家，索要杰勒比结晶只是掩饰。您应该去看看他能提供什么。";
	next;
	if (select("能告诉我怎么去艾尔贝塔吗？:结束对话") == 1) {
		mes "[埃德加]";
		mes "嗯？您可以用双脚走过去。不过如果有钱，我建议您乘船。";
		next;
		if (select("好的，明白了。:但我走累了，而且没钱！") == 1) {
			mes "[埃德加]";
			mes "好的，保重～";
			close;
		}
		mes "[埃德加]";
		mes "好吧……";
		mes "您不想走路又没钱，却还想去那里？哎呀……";
		next;
		mes "[埃德加]";
		mes "好吧好吧。我是船长，可以用低价带您过去。250 Zeny 怎么样？";
		next;
		if (select("好的～！:呸，太贵了！！") == 1) {
			set MISC_QUEST,MISC_QUEST | 16;
			if (Zeny < 250) {
				mes "[埃德加]";
				mes "嗯……";
				mes "这点钱不够。";
				mes "去再赚一些吧。";
				close;
			}
			set Zeny, Zeny-250;
			warp "alberta",195,164;
			end;
		}
		mes "[埃德加]";
		mes "哎呀哎呀，";
		mes "如果您觉得这算敲诈……";
		close;
	}
	mes "[埃德加]";
	mes "好吧，就这样。";
	mes "回头见～";
	close;
}
