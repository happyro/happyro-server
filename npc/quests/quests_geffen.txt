// Number Match Game :: gef_yagu
//============================================================
geffen,122,77,4	script	尼娅#yagu	700,{
	mes "[尼娅]";
	mes "你好，我是尼娅，游戏大师毛翁的第五位弟子。";
	mes "你想和我玩一场数字匹配游戏吗？准备好了吗？";
	next;
	switch(select("什么是数字匹配？:练习数字匹配:正式数字匹配:纪录保持者:取消")) {
	case 1:
		mes "[尼娅]";
		mes "数字匹配游戏的目标是猜出我正在想的数字。";
		mes "当然，我选择数字时也要遵守一些规则。";
		next;
		mes "[尼娅]";
		mes "我想的数字必须是三位数，每一位都不能是0，三位数字也不能重复。";
		next;
		mes "[尼娅]";
		mes "例如，我不能使用103、112、252或701。这些数字要么包含0，要么有重复的数字。";
		mes "现在明白了吗？";
		next;
		mes "[尼娅]";
		mes "正式游戏只有5次猜测机会。";
		mes "每次猜测后，我会分别告诉你：有几位数字正确但位置不对，以及有几位数字和位置都正确。";
		next;
		mes "[尼娅]";
		mes "例如，假设我想的数字是168，你猜的是678。";
		mes "6是数字正确但位置不对，8则是数字和位置都正确。";
		next;
		mes "[尼娅]";
		mes "我建议你先试试练习模式，熟悉之后再玩正式的数字匹配游戏。";
		mes "别忘了，速度也会计入成绩！";
		next;
		mes "[尼娅]";
		mes "如果你在数字匹配游戏中创造新纪录，你的名字就会被记录下来。";
		mes "别错过这个获得名声甚至财富的机会哦～";
		close;
	case 2:
		mes "[尼娅]";
		mes "好，我们来练习数字匹配。请猜猜我的数字，好吗？";
		mes "如果要取消，请输入1000。";
		set .@practice,1;
		break;
	case 3:
		mes "[尼娅]";
		mes "好，我们来玩数字匹配！";
		mes "你有 5 次机会猜出我想的数字。";
		mes "如果要取消，请输入1000。";
		set .@practice,0;
		break;
	case 4:
		set .@name_record$, $050908_minus1_yagu$;
		if (.@name_record$ == "") //custom
			set .@name_record$,"暂无纪录";
		set .@score_record, $050908_minus1_yagu;
		set .@score_min, .@score_record / 60;
		set .@score_sec, .@score_record % 60;
		mes "[尼娅]";
		mes "^ff0000" + .@name_record$ + "^000000是数字匹配游戏的纪录保持者，用时为：";
		mes "^ff0000" + .@score_min + " 分钟、" + .@score_sec + " 秒^000000。";
		close;
	case 5:
		mes "[尼娅]";
		mes "冒险者，请保重。想玩数字匹配游戏时，随时来找我吧。再见～";
		close;
	}
	next;

	// Generate 3 digits of a number.
	while(1) {
		set .@yagu100, rand(1,9);
		set .@yagu10, rand(1,9);
		set .@yagu1, rand(1,9);
		if ((.@yagu100 != .@yagu10) && (.@yagu100 != .@yagu1) && (.@yagu10 != .@yagu1)) {
			// Store the actual number (unofficial).
			set .@yagu, (100*.@yagu100) + (10*.@yagu10) + .@yagu1;
			break;
		}
	}

	// The official time calculation is based on ServerTime,
	// an int HHMMSS equivalent to atoi(gettimestr("%H%M%S",7)).
	// UNIX time is really a better way to do this.
	set .@start_time, gettimetick(2);

	while(1) {

		// Guess a number that meets all conditions.
		while(1) {
			if (input(.@input,100,999) != 0) {
				mes "[尼娅]";
				mes "你取消了数字匹配游戏。";
				close;
			}
			set .@input100, .@input / 100;
			set .@input10, (.@input % 100) / 10;
			set .@input1, .@input % 10;
			if ((.@input100 > 0) && (.@input10 > 0) && (.@input1 > 0)) {
				if ((.@input100 != .@input10) && (.@input100 != .@input1) && (.@input10 != .@input1))
					break;
			}
			mes "[尼娅]";
			mes "你输入的数字包含0，或在不同位置重复使用了相同的数字。";
			mes "请重新猜测。";
			next;
		}

		set .@retry, .@retry+1;
		mes "[尼娅]";
		mes "^ff0000"+strcharinfo(0)+"^000000，你的猜测是 ^0000ff"+.@input+"^000000。";
		mes "请稍等，我来计算结果。";
		next;

		// Correct digits in correct locations.
		set .@strike,0;
		if (.@yagu100 == .@input100)
			set .@strike, .@strike+1;
		if (.@yagu10 == .@input10)
			set .@strike, .@strike+1;
		if (.@yagu1 == .@input1)
			set .@strike, .@strike+1;

		// Correct digits in incorrect locations.
		set .@ball,0;
		if ((.@yagu100 == .@input10) || (.@yagu100 == .@input1))
			set .@ball, .@ball+1;
		if ((.@yagu10 == .@input100) || (.@yagu10 == .@input1))
			set .@ball, .@ball+1;
		if ((.@yagu1 == .@input100) || (.@yagu1 == .@input10))
			set .@ball, .@ball+1;

		if (.@strike == 3) {
			if (.@practice) {
				mes "[尼娅]";
				mes "没错，我想的数字是^ff0000"+.@yagu+"^000000！";
				mes "恭喜你赢得了这场练习赛！";
				mes "现在你已经准备好挑战正式比赛了～";
				close;
			} else {

				// Calculate final time and check against the previous record.
				set .@end_time, gettimetick(2);
				if (.@end_time < .@start_time)
					set .@end_time, .@end_time + 3600;
				set .@total_time, .@end_time - .@start_time;
				set .@total_min, .@total_time / 60;
				set .@total_sec, .@total_time % 60;
				set .@name_record$, $050908_minus1_yagu$;
				if (.@name_record$ == "") //custom
					set .@name_record$,"暂无纪录";
				set .@score_record, $050908_minus1_yagu;
				set .@score_min, .@score_record / 60;
				set .@score_sec, .@score_record % 60;

				if (.@score_record >= .@total_time || !.@score_record) { //custom (second condition)
					set $050908_minus1_yagu$, strcharinfo(0);
					set $050908_minus1_yagu, .@total_time;
					mes "[尼娅]";
					mes "太棒了！你猜中了我的数字，答案是^ff0000" + .@yagu + "^000000。";
					mes "你用时^ff0000" + .@total_min + " 分钟、" + .@total_sec + " 秒^000000，打破了之前^ff0000" + .@score_min + " 分钟、" + .@score_sec + " 秒^000000的纪录。";
					next;
					mes "[尼娅]";
					mes "^ff0000" + .@name_record$ + "^000000创造了那项旧纪录，而现在你成为了数字匹配游戏新的挑战目标。";
					mes "恭喜你创造了新纪录！";
					close;
				}
				mes "[尼娅]";
				mes "做得好，你猜中了我的数字！";
				mes "答案当然是 ^ff0000" + .@yagu + "^000000。";
				mes "你成功猜中这个数字，用时^ff0000" + .@total_min + " 分钟、" + .@total_sec + " 秒^000000。";
				next;
				mes "[尼娅]";
				mes "^ff0000" + .@name_record$ + "^000000是当前数字匹配游戏的纪录保持者，用时^ff0000" + .@score_min + " 分钟、" + .@score_sec + " 秒^000000。";
				mes "祝你好运，希望有一天你能打破这项纪录～";
				close;
			}
		} else {
			if (.@practice) {
				mes "[尼娅]";
				mes "你的猜测中有 ^ff0000" + .@ball + "^000000 个数字正确但位置不对。";
				mes "另有 ^ff0000" + .@strike + "^000000 个数字和位置都正确。";
				mes "好，再试一次吧～";
				mes "";
				next;
			} else {
				mes "[尼娅]";
				mes "你上次的猜测中有 ^ff0000" + .@ball + "^000000 个数字正确但位置不对。";
				mes "另有 ^ff0000" + .@strike + "^000000 个数字和位置都正确。";
				mes "";
				mes "";
				switch(.@retry) {
					case 1: mes "你还剩4次猜测机会。"; break;
					case 2: mes "你还剩3次猜测机会。"; break;
					case 3: mes "你还剩2次猜测机会。"; break;
					case 4: mes "你只剩最后1次猜测机会了……"; break;
					case 5: mes "正确答案是^ff0000" + .@yagu + "^000000。"; close;
				}
				next;
			}
		}
	}
}
