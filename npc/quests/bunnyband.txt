alberta,26,229,0	script	卡普拉员工#bunny	83,{
	if (BUNYBND == 1) {
		mes "[卡普拉员工]";
		mes "你好~！";
		mes "想参加卡普拉";
		mes "公司的特别";
		mes "^529DFF兔兔发带活动^000000吗？";
		mes "";
		next;
		switch(select("好的，我带来了物品。:活动说明:取消")) {
		case 1:
			mes "[卡普拉员工]";
			mes "好的~";
			mes "让我检查一下";
			mes "你是否带齐了";
			mes "所有物品……";
			next;
			setarray .@Items[0],949,100,706,1,722,1,2213,1;
			for(set .@i,0; .@i<8; set .@i,.@i+2)
				if (countitem(.@Items[.@i]) < .@Items[.@i+1]) {
					mes "[卡普拉员工]";
					mes "哦，抱歉，";
					mes "你至少需要";
					mes "带来";
					mes callfunc("F_InsertPlural",.@Items[.@i+1],getitemname(.@Items[.@i]))+".";
					close;
				}
			mes "[卡普拉员工]";
			mes "太好了，我看你已经";
			mes "收集齐了制作兔兔发带";
			mes "所需的全部物品。";
			mes "请稍等片刻，";
			mes "我马上把它组装起来……";
			mes "";
			next;
			for(set .@i,0; .@i<8; set .@i,.@i+2)
				if (countitem(.@Items[.@i]) < .@Items[.@i+1]) {
					mes "[卡普拉员工]";
					mes "嗯？抱歉，";
					mes "我现在确实无法";
					mes "制作这个。你需要";
					mes ""+callfunc("F_InsertPlural",.@Items[.@i+1],getitemname(.@Items[.@i]));
					mes "才能让我组装这个";
					mes "兔兔发带……";
					close;
				}
			delitem 949,100; //Feather
			delitem 706,1; //Four_Leaf_Clover
			delitem 722,1; //Scarlet_Jewel
			delitem 2213,1; //Cat_Hairband
			mes "[卡普拉员工]";
			mes "啊，给你~";
			mes "完美的兔兔发带！";
			mes "希望你喜欢。";
			getitem 2214,1; //Bunny_Band
			set BUNYBND,0;
			next;
			mes "[卡普拉员工]";
			mes "感谢你参加这次";
			mes "特别活动，也感谢你";
			mes "一直以来对";
			mes "卡普拉服务的支持~";
			mes "";
			close;
		case 2:
			goto L_Info;
		case 3:
			close;
		}
	}
	mes "[卡普拉员工]";
	mes "你好~！想参加卡普拉公司举办的";
	mes "特别 ^529DFF兔兔发带活动^000000吗？";
	mes "活动由";
	mes "卡普拉公司主办，";
	mes "艾尔贝塔商人公会赞助。";
	next;
	switch(select("参加活动:活动说明:取消")) {
	case 1:
		set BUNYBND,1;
		mes "[卡普拉员工]";
		mes "太好了！感谢你的参与！";
		mes "如果你还不知道，";
		mes "想让我为你制作兔兔发带，";
		mes "就需要收集这些物品……";
		mes "";
		mes "";
		next;
		mes "[卡普拉员工]";
		mes "100 根羽毛，";
		mes "1 片四叶幸运草，";
		mes "1 个小猫发带和";
		mes "1 颗珍珠。就这些！";
		mes "祝你好运，我会在这里";
		mes "等你回来~";
		close;
	case 2:
		break;
	case 3:
		close;
	}
L_Info:
	mes "[卡普拉员工]";
	mes "为了感谢我们尊贵的";
	mes "顾客，卡普拉公司";
	mes "准备了特别活动：";
	mes "卡普拉员工会为";
	mes "带来所需物品的冒险者";
	mes "制作兔兔发带。";
	next;
	mes "[卡普拉员工]";
	mes "要制作这件特别的";
	mes "独一无二的物品，";
	mes "请带来 100 根羽毛、";
	mes "1 片四叶幸运草、";
	mes "1 个小猫发带和";
	mes "1 颗珍珠。";
	next;
	mes "[卡普拉员工]";
	mes "准备好后，";
	mes "带着这些物品回来找我。";
	mes "（艾尔贝塔商人公会赞助。）";
	mes "";
	mes "";
	close;
}
